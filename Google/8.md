## 1. Router Broadcast and Shutdown

### Problem Statement
Routers can send "Broadcast & Shut Down" messages. When a router receives this message, it broadcasts to all routers within its wireless range, then shuts down. Given router locations (2D coordinates), source, destination, and wireless range, determine if a message can reach the destination.

### Example
```
Input:
routers = {
  A: (0, 0),
  B: (0, 8),
  C: (0, 17),
  D: (11, 0)
}
range = 10
source = A, destination = C

Output: true

Explanation:
- A broadcasts to B (distance = 8 ≤ 10), then shuts down
- B broadcasts to C (distance = 9 ≤ 10), then shuts down
- Message reaches C

source = A, destination = D
Output: false (D is never in range)
```

### C++ Solution
```cpp
class Solution {
public:
    bool canMessageReach(map<string, pair<int, int>>& routers,
                         string source, string destination, int range) {
        if (source == destination) return true;
        
        // BFS to simulate message propagation
        queue<string> q;
        set<string> visited;
        
        q.push(source);
        visited.insert(source);
        
        while (!q.empty()) {
            string current = q.front();
            q.pop();
            
            auto [x1, y1] = routers[current];
            
            // Broadcast to all routers in range
            for (auto& [name, pos] : routers) {
                if (visited.count(name)) continue;
                
                auto [x2, y2] = pos;
                double distance = sqrt(pow(x2 - x1, 2) + pow(y2 - y1, 2));
                
                if (distance <= range) {
                    if (name == destination) return true;
                    
                    visited.insert(name);
                    q.push(name);
                }
            }
        }
        
        return false;
    }
    
    // Helper function to calculate distance
    double getDistance(pair<int, int> p1, pair<int, int> p2) {
        return sqrt(pow(p2.first - p1.first, 2) + 
                   pow(p2.second - p1.second, 2));
    }
};

// Optimized version with integer distance (avoiding floating point)
class SolutionOptimized {
public:
    bool canMessageReach(vector<pair<string, pair<int, int>>>& routers,
                         string source, string destination, int range) {
        map<string, pair<int, int>> routerMap;
        for (auto& [name, pos] : routers) {
            routerMap[name] = pos;
        }
        
        queue<string> q;
        set<string> visited;
        
        q.push(source);
        visited.insert(source);
        
        long long rangeSquared = (long long)range * range;
        
        while (!q.empty()) {
            string current = q.front();
            q.pop();
            
            auto [x1, y1] = routerMap[current];
            
            for (auto& [name, pos] : routerMap) {
                if (visited.count(name)) continue;
                
                auto [x2, y2] = pos;
                long long distSquared = (long long)(x2 - x1) * (x2 - x1) + 
                                       (long long)(y2 - y1) * (y2 - y1);
                
                if (distSquared <= rangeSquared) {
                    if (name == destination) return true;
                    
                    visited.insert(name);
                    q.push(name);
                }
            }
        }
        
        return false;
    }
};
```

## 2. IP Address to City Mapping

### Problem Statement
Given a file with IP ranges and city names, write a function that takes an IP address and returns its associated city.

### Example
```
Input:
File:
1.0.1.1, 1.0.1.10, NYC
1.0.1.20, 1.0.1.30, SF
1.0.1.35, 1.0.1.50, LA

Query: 1.0.1.9
Output: NYC

Query: 1.0.1.25
Output: SF
```

### C++ Solution
```cpp
class IPCityMapper {
private:
    map<long long, pair<long long, string>> ranges; // start IP -> {end IP, city}
    
public:
    long long ipToLong(string ip) {
        long long result = 0;
        stringstream ss(ip);
        string part;
        
        while (getline(ss, part, '.')) {
            result = result * 256 + stoi(part);
        }
        
        return result;
    }
    
    void addRange(string startIP, string endIP, string city) {
        long long start = ipToLong(startIP);
        long long end = ipToLong(endIP);
        ranges[start] = {end, city};
    }
    
    string findCity(string ip) {
        long long ipNum = ipToLong(ip);
        
        // Find the range that could contain this IP
        auto it = ranges.upper_bound(ipNum);
        
        if (it != ranges.begin()) {
            --it;
            auto [start, endCity] = *it;
            auto [end, city] = endCity;
            
            if (ipNum >= start && ipNum <= end) {
                return city;
            }
        }
        
        return "Not Found";
    }
    
    // Alternative: Binary search on sorted ranges
    vector<tuple<long long, long long, string>> sortedRanges;
    
    void buildFromFile(vector<tuple<string, string, string>>& data) {
        for (auto& [startIP, endIP, city] : data) {
            long long start = ipToLong(startIP);
            long long end = ipToLong(endIP);
            sortedRanges.push_back({start, end, city});
        }
        
        sort(sortedRanges.begin(), sortedRanges.end());
    }
    
    string findCityBinarySearch(string ip) {
        long long ipNum = ipToLong(ip);
        
        int left = 0, right = sortedRanges.size() - 1;
        
        while (left <= right) {
            int mid = left + (right - left) / 2;
            auto [start, end, city] = sortedRanges[mid];
            
            if (ipNum >= start && ipNum <= end) {
                return city;
            } else if (ipNum < start) {
                right = mid - 1;
            } else {
                left = mid + 1;
            }
        }
        
        return "Not Found";
    }
};
```

## 3. Valid Organic Molecule

### Problem Statement
Determine if a molecule is valid based on bond requirements:
- Carbon (C): exactly 4 bonds
- Oxygen (O): exactly 2 bonds
- Hydrogen (H): exactly 1 bond

**Follow-up:** Detect if the molecule is disconnected (multiple separate molecules).

### Example
```
Input:
atoms: {1: 'C', 2: 'H', 3: 'H', 4: 'O'}
bonds: [[1, 2], [1, 3], [1, 4], [1, 4]]

Output: true (C has 4 bonds, H atoms have 1, O has 2)

Invalid example (disconnected):
atoms: {1: 'C', 2: 'O', 3: 'O', 4: 'H', 5: 'H'}
bonds: [[1, 2], [1, 2], [1, 3], [1, 3], [4, 5]]
Output: false (two separate molecules: O=C=O and H-H)
```

### C++ Solution
```cpp
class Solution {
private:
    map<char, int> requiredBonds = {
        {'C', 4},
        {'O', 2},
        {'H', 1}
    };
    
public:
    bool isValidMolecule(map<int, char>& atoms, vector<vector<int>>& bonds) {
        // Count bonds for each atom
        map<int, int> bondCount;
        
        for (auto& bond : bonds) {
            bondCount[bond[0]]++;
            bondCount[bond[1]]++;
        }
        
        // Verify each atom has correct number of bonds
        for (auto& [id, atom] : atoms) {
            int required = requiredBonds[atom];
            int actual = bondCount[id];
            
            if (actual != required) {
                return false;
            }
        }
        
        return true;
    }
    
    // Follow-up: Check if molecule is connected
    bool isValidAndConnected(map<int, char>& atoms, vector<vector<int>>& bonds) {
        // First check bond requirements
        if (!isValidMolecule(atoms, bonds)) {
            return false;
        }
        
        // Build adjacency list
        map<int, vector<int>> graph;
        for (auto& [id, atom] : atoms) {
            graph[id] = vector<int>();
        }
        
        for (auto& bond : bonds) {
            graph[bond[0]].push_back(bond[1]);
            graph[bond[1]].push_back(bond[0]);
        }
        
        // Check connectivity using DFS
        set<int> visited;
        int startNode = atoms.begin()->first;
        
        dfs(startNode, graph, visited);
        
        // All atoms should be visited
        return visited.size() == atoms.size();
    }
    
private:
    void dfs(int node, map<int, vector<int>>& graph, set<int>& visited) {
        visited.insert(node);
        
        for (int neighbor : graph[node]) {
            if (visited.find(neighbor) == visited.end()) {
                dfs(neighbor, graph, visited);
            }
        }
    }
};

// Union-Find approach for connectivity
class SolutionUnionFind {
private:
    map<int, int> parent;
    
    int find(int x) {
        if (parent[x] != x) {
            parent[x] = find(parent[x]);
        }
        return parent[x];
    }
    
    void unite(int x, int y) {
        int px = find(x);
        int py = find(y);
        if (px != py) {
            parent[px] = py;
        }
    }
    
public:
    bool isValidAndConnected(map<int, char>& atoms, vector<vector<int>>& bonds) {
        // Initialize union-find
        for (auto& [id, atom] : atoms) {
            parent[id] = id;
        }
        
        // Count bonds
        map<int, int> bondCount;
        for (auto& bond : bonds) {
            bondCount[bond[0]]++;
            bondCount[bond[1]]++;
            unite(bond[0], bond[1]);
        }
        
        // Check bond requirements
        map<char, int> required = {{'C', 4}, {'O', 2}, {'H', 1}};
        for (auto& [id, atom] : atoms) {
            if (bondCount[id] != required[atom]) {
                return false;
            }
        }
        
        // Check connectivity
        int root = find(atoms.begin()->first);
        for (auto& [id, atom] : atoms) {
            if (find(id) != root) {
                return false;
            }
        }
        
        return true;
    }
};
```

## 4. Largest K-Digit Subsequence

### Problem Statement
Given a sequence S of N digits, find a subsequence of K digits such that the number formed is the largest possible.

### Example
```
Input: S = "4902", K = 2
Output: "92"

Input: S = "4902", K = 3
Output: "902"

Input: S = "142857", K = 1
Output: "8"

Input: S = "142857", K = 2
Output: "87"

Input: S = "142857", K = 4
Output: "4857"
```

### C++ Solution
```cpp
class Solution {
public:
    string largestKDigitSubsequence(string S, int K) {
        int n = S.length();
        string result = "";
        int toRemove = n - K;
        
        for (char digit : S) {
            // Remove smaller digits from result if we can afford to
            while (!result.empty() && result.back() < digit && toRemove > 0) {
                result.pop_back();
                toRemove--;
            }
            result += digit;
        }
        
        // Trim to K digits
        return result.substr(0, K);
    }
    
    // Alternative: Greedy approach with explicit tracking
    string largestKDigitGreedy(string S, int K) {
        int n = S.length();
        string result;
        int start = 0;
        
        for (int i = 0; i < K; i++) {
            char maxDigit = '0';
            int maxIdx = start;
            
            // Find the largest digit in the valid range
            // We need at least (K - i - 1) digits after this position
            int end = n - (K - i);
            
            for (int j = start; j <= end; j++) {
                if (S[j] > maxDigit) {
                    maxDigit = S[j];
                    maxIdx = j;
                }
            }
            
            result += maxDigit;
            start = maxIdx + 1;
        }
        
        return result;
    }
};

// Stack-based approach (most optimal)
class SolutionStack {
public:
    string largestKDigitSubsequence(string S, int K) {
        vector<char> stack;
        int toRemove = S.length() - K;
        
        for (char c : S) {
            while (!stack.empty() && stack.back() < c && toRemove > 0) {
                stack.pop_back();
                toRemove--;
            }
            stack.push_back(c);
        }
        
        // Build result from first K characters
        string result;
        for (int i = 0; i < K; i++) {
            result += stack[i];
        }
        
        return result;
    }
};
```

## 5. Password Generation from Keylogger

### Problem Statement
Given a weak keylogger that only records character frequencies (unordered), generate all possible passwords for a brute force attack.

### Example
```
Input: {'a': 4, 'c': 1, 'e': 1, '1': 1}
Output: All permutations of "aaaace1"

Possible passwords include:
"aaaace1", "aaaaec1", "aaa1ace", etc.
Total: 7! / (4! * 1! * 1! * 1!) = 210 permutations
```

### C++ Solution
```cpp
class Solution {
public:
    vector<string> generatePasswords(map<char, int>& keylog) {
        vector<string> result;
        string current = "";
        
        // Build initial string from frequencies
        vector<char> chars;
        for (auto& [ch, count] : keylog) {
            for (int i = 0; i < count; i++) {
                chars.push_back(ch);
            }
        }
        
        // Generate all permutations
        sort(chars.begin(), chars.end());
        
        do {
            string password(chars.begin(), chars.end());
            result.push_back(password);
        } while (next_permutation(chars.begin(), chars.end()));
        
        return result;
    }
    
    // Backtracking approach for large inputs
    vector<string> generatePasswordsBacktrack(map<char, int>& keylog) {
        vector<string> result;
        string current;
        
        backtrack(keylog, current, result);
        return result;
    }
    
private:
    void backtrack(map<char, int>& keylog, string& current, vector<string>& result) {
        // Check if all characters used
        bool allUsed = true;
        for (auto& [ch, count] : keylog) {
            if (count > 0) {
                allUsed = false;
                break;
            }
        }
        
        if (allUsed) {
            result.push_back(current);
            return;
        }
        
        // Try each available character
        for (auto& [ch, count] : keylog) {
            if (count > 0) {
                current += ch;
                keylog[ch]--;
                
                backtrack(keylog, current, result);
                
                current.pop_back();
                keylog[ch]++;
            }
        }
    }
};

// Optimized version to avoid duplicates
class SolutionOptimized {
public:
    vector<string> generatePasswords(map<char, int>& keylog) {
        vector<string> result;
        string current;
        vector<pair<char, int>> freq;
        
        for (auto& p : keylog) {
            freq.push_back(p);
        }
        
        backtrack(freq, current, result);
        return result;
    }
    
private:
    void backtrack(vector<pair<char, int>>& freq, string& current, 
                   vector<string>& result) {
        bool complete = true;
        
        for (auto& [ch, count] : freq) {
            if (count > 0) {
                complete = false;
                break;
            }
        }
        
        if (complete) {
            result.push_back(current);
            return;
        }
        
        for (int i = 0; i < freq.size(); i++) {
            if (freq[i].second > 0) {
                current += freq[i].first;
                freq[i].second--;
                
                backtrack(freq, current, result);
                
                current.pop_back();
                freq[i].second++;
            }
        }
    }
};
```

## 6. Player Ranking from Game Results

### Problem Statement
Given N players and match results between pairs of players, determine which players' ranks can be definitively determined.

### Example
```
Input:
matches = [
  (P1, P2, P1_won),
  (P2, P3, P3_won),
  (P3, P4, P4_won),
  (P1, P4, P4_won)
]

Output: Players whose ranks are definite
```

### C++ Solution
```cpp
class Solution {
public:
    vector<string> findDefiniteRanks(vector<tuple<string, string, string>>& matches) {
        set<string> players;
        map<string, set<string>> beaten; // beaten[A] = set of players A beat
        map<string, set<string>> lostTo; // lostTo[A] = set of players A lost to
        
        // Build graphs
        for (auto& [p1, p2, winner] : matches) {
            players.insert(p1);
            players.insert(p2);
            
            if (winner == p1) {
                beaten[p1].insert(p2);
                lostTo[p2].insert(p1);
            } else {
                beaten[p2].insert(p1);
                lostTo[p1].insert(p2);
            }
        }
        
        int n = players.size();
        vector<string> result;
        
        for (const string& player : players) {
            // Count how many players this player beat (directly or indirectly)
            set<string> canBeat = getReachable(player, beaten);
            // Count how many players beat this player (directly or indirectly)
            set<string> beatenBy = getReachable(player, lostTo);
            
            // If rank is definite: canBeat + beatenBy + 1 (self) == n
            if (canBeat.size() + beatenBy.size() + 1 == n) {
                // Additionally check no conflicts
                bool hasConflict = false;
                for (const string& upper : beatenBy) {
                    for (const string& lower : canBeat) {
                        // Everyone above should beat everyone below
                        if (beaten[upper].find(lower) == beaten[upper].end()) {
                            set<string> upperReach = getReachable(upper, beaten);
                            if (upperReach.find(lower) == upperReach.end()) {
                                hasConflict = true;
                                break;
                            }
                        }
                    }
                    if (hasConflict) break;
                }
                
                if (!hasConflict) {
                    result.push_back(player);
                }
            }
        }
        
        return result;
    }
    
private:
    set<string> getReachable(string start, map<string, set<string>>& graph) {
        set<string> reachable;
        queue<string> q;
        q.push(start);
        
        while (!q.empty()) {
            string curr = q.front();
            q.pop();
            
            for (const string& next : graph[curr]) {
                if (reachable.find(next) == reachable.end()) {
                    reachable.insert(next);
                    q.push(next);
                }
            }
        }
        
        return reachable;
    }
};
```

## 7. Line Sweep with Binary Search

### Problem Statement
Apply line sweep algorithm with binary search optimization (similar to Zero Array Transformation).

### Example
```
Input: Array and range operations
Operations: [(1, 3), (2, 5), (4, 6)]
Each operation decrements range by 1

Output: Minimum operations to make array zero
```

### C++ Solution
```cpp
class Solution {
public:
    int minOperations(vector<int>& nums) {
        int n = nums.size();
        vector<pair<int, int>> events; // (position, delta)
        
        // Line sweep with difference array
        vector<int> diff(n + 1, 0);
        int operations = 0;
        
        for (int i = 0; i < n; i++) {
            diff[i] += (i > 0 ? diff[i - 1] : 0);
            
            if (nums[i] > diff[i]) {
                int needed = nums[i] - diff[i];
                operations += needed;
                diff[i] += needed;
                
                // Mark end of range (example: fixed length k)
                // This is simplified - actual implementation depends on problem
            }
        }
        
        return operations;
    }
    
    // Binary search on answer
    bool canMakeZero(vector<int>& nums, int maxOps, int k) {
        int n = nums.size();
        vector<int> current(nums);
        int ops = 0;
        
        for (int i = 0; i < n; i++) {
            if (current[i] > 0) {
                int needed = current[i];
                if (ops + needed > maxOps) return false;
                
                ops += needed;
                
                // Apply operation to range [i, min(i+k, n))
                for (int j = i; j < min(i + k, n); j++) {
                    current[j] -= needed;
                }
            }
        }
        
        return true;
    }
    
    int minOperationsBinarySearch(vector<int>& nums, int k) {
        int left = 0, right = accumulate(nums.begin(), nums.end(), 0);
        int result = right;
        
        while (left <= right) {
            int mid = left + (right - left) / 2;
            
            if (canMakeZero(nums, mid, k)) {
                result = mid;
                right = mid - 1;
            } else {
                left = mid + 1;
            }
        }
        
        return result;
    }
};
```

## 8. Dice Game Win Count

### Problem Statement
Two players play a dice game. Each die has equal number of sides but different values. Determine how many times Player 1 wins. Solve in O(N²), O(N log N), and O(N).

### Example
```
Input:
dice1 = [1, 2, 3, 4, 5, 6]
dice2 = [2, 2, 3, 3, 4, 4]

Output: Count of outcomes where dice1 > dice2
```

### C++ Solution
```cpp
class Solution {
public:
    // O(N^2) - Brute Force
    int countWinsN2(vector<int>& dice1, vector<int>& dice2) {
        int wins = 0;
        
        for (int val1 : dice1) {
            for (int val2 : dice2) {
                if (val1 > val2) {
                    wins++;
                }
            }
        }
        
        return wins;
    }
    
    // O(N log N) - Sort and Binary Search
    int countWinsNLogN(vector<int>& dice1, vector<int>& dice2) {
        sort(dice2.begin(), dice2.end());
        int wins = 0;
        
        for (int val : dice1) {
            // Count how many values in dice2 are less than val
            int count = lower_bound(dice2.begin(), dice2.end(), val) - dice2.begin();
            wins += count;
        }
        
        return wins;
    }
    
    // O(N) - Using counting/frequency
    int countWinsN(vector<int>& dice1, vector<int>& dice2) {
        // Assuming dice values are in a limited range
        map<int, int> freq2;
        
        for (int val : dice2) {
            freq2[val]++;
        }
        
        int wins = 0;
        
        for (int val1 : dice1) {
            for (auto& [val2, count] : freq2) {
                if (val1 > val2) {
                    wins += count;
                }
            }
        }
        
        return wins;
    }
    
    // O(N) - Optimal with prefix sum
    int countWinsOptimal(vector<int>& dice1, vector<int>& dice2) {
        int maxVal = max(*max_element(dice1.begin(), dice1.end()),
                         *max_element(dice2.begin(), dice2.end()));
        
        vector<int> freq(maxVal + 1, 0);
        
        // Count frequencies in dice2
        for (int val : dice2) {
            freq[val]++;
        }
        
        // Build prefix sum
        for (int i = 1; i <= maxVal; i++) {
            freq[i] += freq[i - 1];
        }
        
        int wins = 0;
        
        // For each value in dice1, count values in dice2 that are smaller
        for (int val : dice1) {
            if (val > 0) {
                wins += freq[val - 1];
            }
        }
        
        return wins;
    }
};
```

## 9. Magical String Pairs Triplets

### Problem Statement
Find the number of triplets (i, j, k) where s1 = s[i..j] and s2 = s[j+1..k] have equal number of distinct characters.

### Example
```
Input: s = "abcabc"
Output: Count of valid triplets

For i=0, j=2, k=5:
s1 = "abc" (3 distinct)
s2 = "abc" (3 distinct)
Valid triplet
```

### C++ Solution
```cpp
class Solution {
public:
    int countMagicalTriplets(string s) {
        int n = s.length();
        int count = 0;
        
        // Brute force: O(n^3)
        for (int i = 0; i < n; i++) {
            for (int j = i; j < n - 1; j++) {
                for (int k = j + 1; k < n; k++) {
                    int distinct1 = countDistinct(s, i, j);
                    int distinct2 = countDistinct(s, j + 1, k);
                    
                    if (distinct1 == distinct2) {
                        count++;
                    }
                }
            }
        }
        
        return count;
    }
    
private:
    int countDistinct(string& s, int start, int end) {
        set<char> unique;
        for (int i = start; i <= end; i++) {
            unique.insert(s[i]);
        }
        return unique.size();
    }
    
public:
    // Optimized with memoization
    int countMagicalTripletsOptimized(string s) {
        int n = s.length();
        
        // Precompute distinct counts for all substrings
        vector<vector<int>> distinct(n, vector<int>(n, 0));
        
        for (int i = 0; i < n; i++) {
            set<char> unique;
            for (int j = i; j < n; j++) {
                unique.insert(s[j]);
                distinct[i][j] = unique.size();
            }
        }
        
        int count = 0;
        
        for (int i = 0; i < n; i++) {
            for (int j = i; j < n - 1; j++) {
                for (int k = j + 1; k < n; k++) {
                    if (distinct[i][j] == distinct[j + 1][k]) {
                        count++;
                    }
                }
            }
        }
        
        return count;
    }
    
    // Using bitmask for distinct characters
    int countMagicalTripletsBitmask(string s) {
        int n = s.length();
        int count = 0;
        
        for (int i = 0; i < n; i++) {
            int mask1 = 0;
            
            for (int j = i; j < n - 1; j++) {
                mask1 |= (1 << (s[j] - 'a'));
                int bits1 = __builtin_popcount(mask1);
                
                int mask2 = 0;
                for (int k = j + 1; k < n; k++) {
                    mask2 |= (1 << (s[k] - 'a'));
                    int bits2 = __builtin_popcount(mask2);
                    
                    if (bits1 == bits2) {
                        count++;
                    }
                }
            }
        }
        
        return count;
    }
};
```

## 10. Expression String Evaluation

### Problem Statement
Evaluate a string expression with nested function calls: add, sub, mul, pow, div.

### Example
```
Input: "add(5, mul(2, pow(5,2)))"
Output: 5 + (2 * 25) = 5 + 50 = 55
```

### C++ Solution
```cpp
class Solution {
public:
    int evaluate(string expr) {
        int pos = 0;
        return parseExpression(expr, pos);
    }
    
private:
    int parseExpression(string& expr, int& pos) {
        // Skip whitespace
        while (pos < expr.size() && expr[pos] == ' ') pos++;
        
        if (isdigit(expr[pos]) || expr[pos] == '-') {
            return parseNumber(expr, pos);
        }
        
        // Parse function name
        string func;
        while (pos < expr.size() && isalpha(expr[pos])) {
            func += expr[pos++];
        }
        
        // Skip '('
        pos++;
        
        // Parse arguments
        vector<int> args;
        while (expr[pos] != ')') {
            while (pos < expr.size() && expr[pos] == ' ') pos++;
            
            args.push_back(parseExpression(expr, pos));
            
            while (pos < expr.size() && (expr[pos] == ' ' || expr[pos] == ',')) pos++;
        }
        
        // Skip ')'
        pos++;
        
        return applyFunction(func, args);
    }
    
    int parseNumber(string& expr, int& pos) {
        int sign = 1;
        if (expr[pos] == '-') {
            sign = -1;
            pos++;
        }
        
        int num = 0;
        while (pos < expr.size() && isdigit(expr[pos])) {
            num = num * 10 + (expr[pos] - '0');
            pos++;
        }
        
        return sign * num;
    }
    
    int applyFunction(string func, vector<int>& args) {
        if (func == "add") {
            return args[0] + args[1];
        } else if (func == "sub") {
            return args[0] - args[1];
        } else if (func == "mul") {
            return args[0] * args[1];
        } else if (func == "div") {
            return args[0] / args[1];
        } else if (func == "pow") {
            return pow(args[0], args[1]);
        }
        return 0;
    }
};
```

## 11. Insert Range and Query Point

### Problem Statement
Implement two functions:
1. `insertRange(start, end)`: Insert interval [start, end) (end exclusive)
2. `query(point)`: Return true if point is in any interval

### Example
```
insertRange(2, 5)
insertRange(2, 3)
insertRange(9, 13)

Final intervals: [2, 5), [9, 13)

query(0) → false
query(2) → true
query(10) → true
query(5) → false
```

### C++ Solution
```cpp
class RangeManager {
private:
    map<int, int> intervals; // start → end
    
public:
    void insertRange(int start, int end) {
        if (start >= end) return;
        
        // Find overlapping intervals
        auto it = intervals.upper_bound(start);
        
        if (it != intervals.begin()) {
            --it;
            if (it->second < start) {
                ++it;
            }
        }
        
        // Merge overlapping intervals
        int newStart = start;
        int newEnd = end;
        
        while (it != intervals.end() && it->first <= end) {
            newStart = min(newStart, it->first);
            newEnd = max(newEnd, it->second);
            it = intervals.erase(it);
        }
        
        intervals[newStart] = newEnd;
    }
    
    bool query(int point) {
        auto it = intervals.upper_bound(point);
        
        if (it != intervals.begin()) {
            --it;
            return point >= it->first && point < it->second;
        }
        
        return false;
    }
};

// Alternative: Using set of intervals
class RangeManagerSet {
private:
    struct Interval {
        int start, end;
        
        bool operator<(const Interval& other) const {
            return start < other.start;
        }
    };
    
    set<Interval> intervals;
    
public:
    void insertRange(int start, int end) {
        Interval newInterval = {start, end};
        
        auto it = intervals.lower_bound(newInterval);
        
        // Check previous interval for overlap
        if (it != intervals.begin()) {
            auto prev = std::prev(it);
            if (prev->end >= start) {
                newInterval.start = prev->start;
                newInterval.end = max(newInterval.end, prev->end);
                intervals.erase(prev);
            }
        }
        
        // Merge with following intervals
        while (it != intervals.end() && it->start <= newInterval.end) {
            newInterval.end = max(newInterval.end, it->end);
            it = intervals.erase(it);
        }
        
        intervals.insert(newInterval);
    }
    
    bool query(int point) {
        Interval search = {point, point + 1};
        auto it = intervals.upper_bound(search);
        
        if (it != intervals.begin()) {
            --it;
            return point >= it->start && point < it->end;
        }
        
        return false;
    }
};
```

```cpp
        // dp[i][j] = max length ending at i with j decreases used
        vector<vector<int>> dp(n, vector<int>(k + 1, 1));
        
        for (int i = 1; i < n; i++) {
            for (int j = 0; j < i; j++) {
                if (nums[i] > nums[j]) {
                    // Increasing
                    for (int d = 0; d <= k; d++) {
                        dp[i][d] = max(dp[i][d], dp[j][d] + 1);
                    }
```
