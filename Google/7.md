## 1. First Missing Positive

**Problem Link:** [LeetCode 41](https://leetcode.com/problems/first-missing-positive/)

### Problem Statement
Given an unsorted integer array, find the smallest missing positive integer.

### Example
```
Input: nums = [1, 2, 0]
Output: 3

Input: nums = [3, 4, -1, 1]
Output: 2

Input: nums = [7, 8, 9, 11, 12]
Output: 1
```

### C++ Solution
```cpp
class Solution {
public:
    int firstMissingPositive(vector<int>& nums) {
        int n = nums.size();

        // Place each number in its correct position (index = value - 1)
        for (int i = 0; i < n; i++) {
            while (nums[i] > 0 && nums[i] <= n &&
                   nums[nums[i] - 1] != nums[i]) {
                swap(nums[i], nums[nums[i] - 1]);
            }
        }

        // Find the first missing positive
        for (int i = 0; i < n; i++) {
            if (nums[i] != i + 1) {
                return i + 1;
            }
        }

        return n + 1;
    }
};
```

## 2. Task Scheduling with Priority and Duration

### Problem Statement
Schedule tasks with given priorities and time durations. Tasks can only be executed after their duration requirement is met. Design a system to execute tasks concurrently based on priority.

### Example
```
Input:
tasks = [
  {id: 1, priority: 5, duration: 10},
  {id: 2, priority: 3, duration: 5},
  {id: 3, priority: 8, duration: 15}
]

Output: Execution order based on priority: [3, 1, 2]
```

### C++ Solution
```cpp
struct Task {
    int id;
    int priority;
    int duration;
    int startTime;

    Task(int i, int p, int d, int s = 0)
        : id(i), priority(p), duration(d), startTime(s) {}
};

class TaskScheduler {
private:
    struct ComparePriority {
        bool operator()(const Task& a, const Task& b) {
            return a.priority < b.priority; // Max heap
        }
    };

    priority_queue<Task, vector<Task>, ComparePriority> readyQueue;
    queue<Task> waitingQueue;
    int currentTime;

public:
    TaskScheduler() : currentTime(0) {}

    void addTask(Task task) {
        if (currentTime >= task.duration) {
            readyQueue.push(task);
        } else {
            waitingQueue.push(task);
        }
    }

    vector<int> executeTasks(vector<Task>& tasks) {
        vector<int> executionOrder;

        for (auto& task : tasks) {
            addTask(task);
        }

        while (!readyQueue.empty() || !waitingQueue.empty()) {
            // Move tasks from waiting to ready
            queue<Task> tempQueue;
            while (!waitingQueue.empty()) {
                Task task = waitingQueue.front();
                waitingQueue.pop();

                if (currentTime >= task.duration) {
                    readyQueue.push(task);
                } else {
                    tempQueue.push(task);
                }
            }
            waitingQueue = tempQueue;

            // Execute highest priority task
            if (!readyQueue.empty()) {
                Task task = readyQueue.top();
                readyQueue.pop();

                executionOrder.push_back(task.id);
                currentTime++;
            } else {
                currentTime++;
            }
        }

        return executionOrder;
    }
};

// Thread pool executor version
class ThreadPoolTaskScheduler {
private:
    priority_queue<Task, vector<Task>, TaskScheduler::ComparePriority> taskQueue;
    int numThreads;

public:
    ThreadPoolTaskScheduler(int threads) : numThreads(threads) {}

    vector<int> schedule(vector<Task>& tasks) {
        vector<int> result;

        // Sort by priority
        sort(tasks.begin(), tasks.end(),
             [](const Task& a, const Task& b) {
                 return a.priority > b.priority;
             });

        for (const auto& task : tasks) {
            result.push_back(task.id);
        }

        return result;
    }
};
```

## 3. Formula Parentheses Simplification

### Problem Statement
Given a mathematical formula with parentheses and variables, remove all parentheses and simplify the expression.

### Example
```
Input: "a - (a - b)"
Output: "b"

Input: "a - (b - c)"
Output: "a - b + c"

Input: "a - (b - (c - d))"
Output: "a - b + c - d"
```

### C++ Solution
```cpp
class Solution {
public:
    string simplifyFormula(string formula) {
        stack<int> signs; // Stack to keep track of signs
        signs.push(1); // Start with positive

        int currentSign = 1;
        map<string, int> coefficients;

        int i = 0;
        while (i < formula.size()) {
            if (formula[i] == ' ') {
                i++;
                continue;
            }

            if (formula[i] == '+') {
                currentSign = signs.top();
                i++;
            } else if (formula[i] == '-') {
                currentSign = -signs.top();
                i++;
            } else if (formula[i] == '(') {
                signs.push(currentSign);
                currentSign = signs.top();
                i++;
            } else if (formula[i] == ')') {
                signs.pop();
                i++;
            } else if (isalpha(formula[i])) {
                // Parse variable
                string var;
                while (i < formula.size() && isalnum(formula[i])) {
                    var += formula[i];
                    i++;
                }
                coefficients[var] += currentSign;
                currentSign = signs.top();
            }
        }

        // Build result
        string result;
        bool first = true;

        for (auto& [var, coef] : coefficients) {
            if (coef == 0) continue;

            if (!first && coef > 0) {
                result += " + ";
            } else if (coef < 0) {
                result += first ? "-" : " - ";
                coef = -coef;
            } else if (!first) {
                result += " + ";
            }

            if (abs(coef) != 1) {
                result += to_string(abs(coef));
            }
            result += var;
            first = false;
        }

        return result.empty() ? "0" : result;
    }
};
```

## 4. Maximum Distance Travel with Energy Constraint

### Problem Statement
Given maximum initial energy K and an array A of wind speeds for N days, find the maximum distance you can travel. Each day you can either:
- Travel: move A[i] distance, energy decreases by 1
- Rest: stay put, energy increases by 1 (max K)

Energy cannot go negative.

### Example
```
Input: K = 5, A = [10, 5, 7, 3, 8, 6]
Output: 39

Explanation: Travel on days with high wind speeds
```

### C++ Solution
```cpp
class Solution {
public:
    int maxDistance(int K, vector<int>& A) {
        int n = A.size();
        // dp[i][e] = max distance after day i with energy e
        vector<vector<int>> dp(n + 1, vector<int>(K + 1, -1));
        dp[0][K] = 0; // Start with max energy, 0 distance

        for (int i = 0; i < n; i++) {
            for (int e = 0; e <= K; e++) {
                if (dp[i][e] == -1) continue;

                // Option 1: Travel (if we have energy)
                if (e > 0) {
                    dp[i + 1][e - 1] = max(dp[i + 1][e - 1],
                                            dp[i][e] + A[i]);
                }

                // Option 2: Rest (energy increases, capped at K)
                int newEnergy = min(e + 1, K);
                dp[i + 1][newEnergy] = max(dp[i + 1][newEnergy],
                                            dp[i][e]);
            }
        }

        // Find maximum distance with any energy level
        int maxDist = 0;
        for (int e = 0; e <= K; e++) {
            if (dp[n][e] != -1) {
                maxDist = max(maxDist, dp[n][e]);
            }
        }

        return maxDist;
    }

    // Space optimized version
    int maxDistanceOptimized(int K, vector<int>& A) {
        vector<int> dp(K + 1, -1);
        dp[K] = 0;

        for (int wind : A) {
            vector<int> newDp(K + 1, -1);

            for (int e = 0; e <= K; e++) {
                if (dp[e] == -1) continue;

                // Travel
                if (e > 0) {
                    newDp[e - 1] = max(newDp[e - 1], dp[e] + wind);
                }

                // Rest
                int newEnergy = min(e + 1, K);
                newDp[newEnergy] = max(newDp[newEnergy], dp[e]);
            }

            dp = newDp;
        }

        return *max_element(dp.begin(), dp.end());
    }
};
```

## 5. Horizontal Axis Dividing Square Areas

### Problem Statement
Given squares in a 2D plane with their top-left coordinates and edge lengths, find a horizontal axis (y-coordinate) that divides the total area of all squares into two equal halves.

### Example
```
Input:
topLeft = [(1, 2), (2, 4)]
sizes = [5, 10]

Square 1: top-left (1, 2), size 5 → covers y ∈ [-3, 2]
Square 2: top-left (2, 4), size 10 → covers y ∈ [-6, 4]

Output: y-coordinate that divides total area equally
```

### C++ Solution
```cpp
class Solution {
public:
    double findDividingAxis(vector<pair<int, int>>& topLeft, vector<int>& sizes) {
        int n = topLeft.size();
        vector<pair<double, double>> events; // (y-coordinate, area change)
        double totalArea = 0;

        for (int i = 0; i < n; i++) {
            int x = topLeft[i].first;
            int y = topLeft[i].second;
            int size = sizes[i];
            double area = (double)size * size;
            totalArea += area;

            // Top edge at y, bottom edge at y - size
            events.push_back({y, area});
            events.push_back({y - size, -area});
        }

        // Sort events by y-coordinate
        sort(events.begin(), events.end());

        double targetArea = totalArea / 2.0;
        double currentAreaAbove = 0;

        for (int i = 0; i < events.size() - 1; i++) {
            double y1 = events[i].first;
            double y2 = events[i + 1].first;

            currentAreaAbove += events[i].second;

            // Binary search within this segment
            if (currentAreaAbove <= targetArea) {
                double areaBelow = totalArea - currentAreaAbove;

                // If target is in this range
                if (areaBelow >= targetArea) {
                    // Need to find exact y in range [y1, y2]
                    // For uniform distribution within segment
                    return y1; // Approximation
                }
            }
        }

        return 0.0;
    }

    // Binary search approach
    double findDividingAxisBinarySearch(vector<pair<int, int>>& topLeft,
                                         vector<int>& sizes) {
        double minY = INT_MAX, maxY = INT_MIN;
        double totalArea = 0;

        // Find range and total area
        for (int i = 0; i < topLeft.size(); i++) {
            int y = topLeft[i].second;
            int size = sizes[i];
            maxY = max(maxY, (double)y);
            minY = min(minY, (double)(y - size));
            totalArea += (double)size * size;
        }

        double left = minY, right = maxY;
        double targetArea = totalArea / 2.0;

        // Binary search for the dividing line
        while (right - left > 1e-6) {
            double mid = (left + right) / 2.0;
            double areaAbove = calculateAreaAbove(topLeft, sizes, mid);

            if (areaAbove < targetArea) {
                right = mid;
            } else {
                left = mid;
            }
        }

        return (left + right) / 2.0;
    }

private:
    double calculateAreaAbove(vector<pair<int, int>>& topLeft,
                              vector<int>& sizes, double y) {
        double area = 0;

        for (int i = 0; i < topLeft.size(); i++) {
            int topY = topLeft[i].second;
            int bottomY = topY - sizes[i];
            int size = sizes[i];

            if (y <= bottomY) {
                // Entire square is above
                area += (double)size * size;
            } else if (y < topY) {
                // Partial square
                double height = topY - y;
                area += height * size;
            }
            // If y >= topY, square is entirely below
        }

        return area;
    }
};
```

## 6. Highest Score Anagram

### Problem Statement
Given a dictionary of words, a hash function for words (same for anagrams), and a score function (sum of differences between adjacent characters), find the anagram of a given word with the highest score.

### Example
```
Input:
words = ["cat", "tac", "act", "dog", "god"]
word = "cat"

Scores:
"cat": |c-a| + |a-t| = 2 + 19 = 21
"tac": |t-a| + |a-c| = 19 + 2 = 21
"act": |a-c| + |c-t| = 2 + 17 = 19

Output: "cat" or "tac" (both have highest score 21)
```

### C++ Solution
```cpp
class Solution {
private:
    unordered_map<string, vector<pair<string, int>>> hashToWords;

public:
    string getHash(string word) {
        string sorted = word;
        sort(sorted.begin(), sorted.end());
        return sorted;
    }

    int getScore(string word) {
        int score = 0;
        for (int i = 1; i < word.size(); i++) {
            score += abs(word[i] - word[i - 1]);
        }
        return score;
    }

    void preprocess(vector<string>& words) {
        for (const string& word : words) {
            string hash = getHash(word);
            int score = getScore(word);
            hashToWords[hash].push_back({word, score});
        }

        // Sort each hash group by score (descending)
        for (auto& [hash, wordList] : hashToWords) {
            sort(wordList.begin(), wordList.end(),
                 [](const pair<string, int>& a, const pair<string, int>& b) {
                     return a.second > b.second;
                 });
        }
    }

    string findAnagram(string word) {
        string hash = getHash(word);

        if (hashToWords.find(hash) == hashToWords.end()) {
            return "";
        }

        auto& anagrams = hashToWords[hash];

        // Return highest score anagram that's not the input word
        for (auto& [anagram, score] : anagrams) {
            if (anagram != word) {
                return anagram;
            }
        }

        // If all anagrams are same as word, return second highest
        if (anagrams.size() > 1) {
            return anagrams[1].first;
        }

        return "";
    }
};
```

## 7. Longest Path in DAG

### Problem Statement
Given a Directed Acyclic Graph (DAG), find the maximum path distance from any node to any other node.

### Example
```
Input:
graph = {
  0: [1, 2],
  1: [3],
  2: [3],
  3: []
}

Output: 2 (path: 0 → 1 → 3 or 0 → 2 → 3)
```

### C++ Solution
```cpp
class Solution {
public:
    int longestPath(vector<vector<int>>& graph) {
        int n = graph.size();
        vector<int> dist(n, -1);
        int maxPath = 0;

        for (int i = 0; i < n; i++) {
            if (dist[i] == -1) {
                maxPath = max(maxPath, dfs(i, graph, dist));
            }
        }

        return maxPath;
    }

private:
    int dfs(int node, vector<vector<int>>& graph, vector<int>& dist) {
        if (dist[node] != -1) {
            return dist[node];
        }

        int maxDist = 0;
        for (int next : graph[node]) {
            maxDist = max(maxDist, 1 + dfs(next, graph, dist));
        }

        return dist[node] = maxDist;
    }
};

// Alternative: Topological sort approach
class SolutionTopSort {
public:
    int longestPath(vector<vector<int>>& graph) {
        int n = graph.size();
        vector<int> indegree(n, 0);
        vector<int> dist(n, 0);

        // Calculate indegrees
        for (int i = 0; i < n; i++) {
            for (int next : graph[i]) {
                indegree[next]++;
            }
        }

        // Initialize queue with nodes having 0 indegree
        queue<int> q;
        for (int i = 0; i < n; i++) {
            if (indegree[i] == 0) {
                q.push(i);
            }
        }

        // Process nodes in topological order
        while (!q.empty()) {
            int node = q.front();
            q.pop();

            for (int next : graph[node]) {
                dist[next] = max(dist[next], dist[node] + 1);
                indegree[next]--;

                if (indegree[next] == 0) {
                    q.push(next);
                }
            }
        }

        return *max_element(dist.begin(), dist.end());
    }
};
```

## 8. Modified Dijkstra with Obstacles

### Problem Statement
Find the shortest path in a graph where you can remove up to K obstacles. Each edge may have an obstacle that can be removed to use that path.

### Example
```
Input:
graph with obstacles
start = 0, end = 4, k = 1

Output: Shortest path removing at most 1 obstacle
```

### C++ Solution
```cpp
class Solution {
public:
    int shortestPathWithObstacles(vector<vector<pair<int, int>>>& graph,
                                   vector<bool>& hasObstacle,
                                   int start, int end, int k) {
        int n = graph.size();
        // dist[node][obstacles_removed]
        vector<vector<int>> dist(n, vector<int>(k + 1, INT_MAX));

        priority_queue<tuple<int, int, int>,
                      vector<tuple<int, int, int>>,
                      greater<>> pq;

        pq.push({0, start, 0}); // {distance, node, obstacles_removed}
        dist[start][0] = 0;

        while (!pq.empty()) {
            auto [d, node, removed] = pq.top();
            pq.pop();

            if (node == end) return d;
            if (d > dist[node][removed]) continue;

            for (auto [next, weight] : graph[node]) {
                bool obstacle = hasObstacle[next];
                int newRemoved = removed + (obstacle ? 1 : 0);

                if (newRemoved <= k) {
                    int newDist = d + weight;

                    if (newDist < dist[next][newRemoved]) {
                        dist[next][newRemoved] = newDist;
                        pq.push({newDist, next, newRemoved});
                    }
                }
            }
        }

        int result = INT_MAX;
        for (int i = 0; i <= k; i++) {
            result = min(result, dist[end][i]);
        }

        return result == INT_MAX ? -1 : result;
    }
};
```

## 9. Two Diseases Spread Simulation

### Problem Statement
Similar to rotten oranges but with two diseases starting from different points. Disease 1 is more powerful and blocks disease 2. Determine if disease 2 reaches the destination.

### Example
```
Input:
grid with disease1 start, disease2 start, destination
disease1 spreads and blocks disease2

Output: true/false if disease2 reaches destination
```

### C++ Solution
```cpp
class Solution {
public:
    bool canDiseaseReachDestination(vector<vector<int>>& grid,
                                     pair<int, int> disease1Start,
                                     pair<int, int> disease2Start,
                                     pair<int, int> destination) {
        int m = grid.size(), n = grid[0].size();
        vector<vector<int>> time1(m, vector<int>(n, INT_MAX));
        vector<vector<int>> time2(m, vector<int>(n, INT_MAX));

        // BFS for disease 1
        queue<pair<int, int>> q1;
        q1.push(disease1Start);
        time1[disease1Start.first][disease1Start.second] = 0;

        int dirs[4][2] = {{0,1}, {1,0}, {0,-1}, {-1,0}};

        while (!q1.empty()) {
            auto [x, y] = q1.front();
            q1.pop();

            for (auto& dir : dirs) {
                int nx = x + dir[0];
                int ny = y + dir[1];

                if (nx >= 0 && nx < m && ny >= 0 && ny < n &&
                    grid[nx][ny] != -1 &&
                    time1[nx][ny] > time1[x][y] + 1) {
                    time1[nx][ny] = time1[x][y] + 1;
                    q1.push({nx, ny});
                }
            }
        }

        // BFS for disease 2 (avoiding disease 1)
        queue<pair<int, int>> q2;
        q2.push(disease2Start);
        time2[disease2Start.first][disease2Start.second] = 0;

        while (!q2.empty()) {
            auto [x, y] = q2.front();
            q2.pop();

            if (x == destination.first && y == destination.second) {
                return true;
            }

            for (auto& dir : dirs) {
                int nx = x + dir[0];
                int ny = y + dir[1];

                if (nx >= 0 && nx < m && ny >= 0 && ny < n &&
                    grid[nx][ny] != -1 &&
                    time2[nx][ny] > time2[x][y] + 1 &&
                    time1[nx][ny] > time2[x][y] + 1) { // Disease1 hasn't reached
                    time2[nx][ny] = time2[x][y] + 1;
                    q2.push({nx, ny});
                }
            }
        }

        return false;
    }
};
```

## 10. Employee Availability Timeline

### Problem Statement
Given arrival and departure timings of employees, print the availability status for each time interval showing which employees are present.

### Example
```
Input:
E1: (10, 20)
E2: (15, 45)
E3: (35, 70)

Output:
(10-15) → E1
(15-20) → E1, E2
(20-35) → E2
(35-45) → E2, E3
(45-70) → E3
```

### C++ Solution
```cpp
class Solution {
public:
    vector<tuple<int, int, vector<string>>> getAvailability(
        vector<pair<string, pair<int, int>>>& employees) {

        vector<tuple<int, int, vector<string>>> result;
        map<int, vector<pair<string, int>>> events; // time → {employee, +1/-1}

        for (auto& [name, times] : employees) {
            events[times.first].push_back({name, 1});  // Arrival
            events[times.second].push_back({name, -1}); // Departure
        }

        set<string> current;
        int prevTime = -1;

        for (auto& [time, empEvents] : events) {
            // Output interval before this event
            if (!current.empty() && prevTime != -1) {
                vector<string> emps(current.begin(), current.end());
                result.push_back({prevTime, time, emps});
            }

            // Process events at this time
            for (auto& [emp, delta] : empEvents) {
                if (delta == 1) {
                    current.insert(emp);
                } else {
                    current.erase(emp);
                }
            }

            prevTime = time;
        }

        return result;
    }

    void printAvailability(vector<pair<string, pair<int, int>>>& employees) {
        auto intervals = getAvailability(employees);

        for (auto& [start, end, emps] : intervals) {
            cout << "(" << start << "-" << end << ") → ";
            for (int i = 0; i < emps.size(); i++) {
                cout << emps[i];
                if (i < emps.size() - 1) cout << ", ";
            }
            cout << endl;
        }
    }
};
```

## 11. Peak in Mountain Array with Duplicates

### Problem Statement
Find a peak element in a mountain array that may contain duplicates. A peak is an element greater than its neighbors.

### Example
```
Input: arr = [1, 2, 3, 3, 3, 2, 1]
Output: 2 (index where arr[i] = 3)

Input: arr = [1, 2, 1, 3, 5, 6, 4]
Output: 5 (index where arr[i] = 6)
```

### C++ Solution
```cpp
class Solution {
public:
    int findPeakElement(vector<int>& nums) {
        int n = nums.size();
        if (n == 1) return 0;

        int left = 0, right = n - 1;

        while (left < right) {
            int mid = left + (right - left) / 2;

            // Handle duplicates
            if (mid > 0 && nums[mid] == nums[mid - 1]) {
                // Expand to find different element
                int i = mid;
                while (i > left && nums[i] == nums[mid]) i--;
                int j = mid;
                while (j < right && nums[j] == nums[mid]) j++;

                if (i >= left && nums[i] > nums[mid]) {
                    right = mid - 1;
                } else if (j <= right && nums[j] > nums[mid]) {
                    left = mid + 1;
                } else {
                    // nums[mid] might be peak
                    return mid;
                }
            } else if (nums[mid] < nums[mid + 1]) {
                left = mid + 1;
            } else {
                right = mid;
            }
        }

        return left;
    }

    // Linear scan fallback for heavy duplicates
    int findPeakLinear(vector<int>& nums) {
        for (int i = 0; i < nums.size(); i++) {
            bool isPeak = true;

            if (i > 0 && nums[i] <= nums[i - 1]) isPeak = false;
            if (i < nums.size() - 1 && nums[i] <= nums[i + 1]) isPeak = false;

            if (isPeak) return i;
        }

        return 0;
    }
};
```

## 12. Binary Tree Cameras

**Problem Link:** [LeetCode 968](https://leetcode.com/problems/binary-tree-cameras/)

### Problem Statement
Install cameras on tree nodes to monitor all nodes. A camera at a node monitors its parent, itself, and immediate children. Find minimum cameras needed.

### Example
```
Input: root = [0,0,null,0,0]
Output: 1

Input: root = [0,0,null,0,null,0,null,null,0]
Output: 2
```

### C++ Solution
```cpp
class Solution {
private:
    int cameras = 0;

    // States: 0 = needs camera, 1 = has camera, 2 = covered
    enum State { NEEDS_CAMERA = 0, HAS_CAMERA = 1, COVERED = 2 };

public:
    int minCameraCover(TreeNode* root) {
        if (dfs(root) == NEEDS_CAMERA) {
            cameras++;
        }
        return cameras;
    }

private:
    int dfs(TreeNode* node) {
        if (!node) return COVERED;

        int left = dfs(node->left);
        int right = dfs(node->right);

        // If any child needs camera, place camera here
        if (left == NEEDS_CAMERA || right == NEEDS_CAMERA) {
            cameras++;
            return HAS_CAMERA;
        }

        // If any child has camera, this node is covered
        if (left == HAS_CAMERA || right == HAS_CAMERA) {
            return COVERED;
        }

        // Both children are covered, this node needs camera
        return NEEDS_CAMERA;
    }
};
```

## 13. Minimum Weight to Disconnect Leaf Nodes

### Problem Statement
Given a tree where each edge has a weight, find the minimum total weight of edges to cut to disconnect all leaf nodes from the tree.

### Example
```
Input:
Tree structure:
    A
   / \
  B   C
 / \
D   E

Weights: A-B: 2, A-C: 4, B-D: 2, B-E: 1

Output: 6
Explanation: Cut B (weight 2) and C (weight 4) → 2 + 4 = 6
```

### C++ Solution
```cpp
class Solution {
public:
    int minWeightToDisconnectLeaves(int root,
                                     unordered_map<int, vector<pair<int, int>>>& tree) {
        unordered_set<int> visited;
        return dfs(root, -1, tree, visited);
    }

private:
    int dfs(int node, int parent,
            unordered_map<int, vector<pair<int, int>>>& tree,
            unordered_set<int>& visited) {

        visited.insert(node);

        // Check if leaf node
        bool isLeaf = true;
        int totalWeight = 0;

        for (auto& [child, weight] : tree[node]) {
            if (child == parent || visited.count(child)) continue;

            isLeaf = false;
            int childWeight = dfs(child, node, tree, visited);

            // If child subtree has leaves, we need to cut
            if (childWeight >= 0) {
                totalWeight += min(weight, childWeight);
            }
        }

        // If this is a leaf, return -1 to signal parent to cut
        if (isLeaf && node != parent) {
            return -1;
        }

        return totalWeight;
    }
};

// Alternative greedy approach
class SolutionGreedy {
public:
    int minWeightToDisconnectLeaves(int root,
                                     unordered_map<int, vector<pair<int, int>>>& tree) {
        int totalWeight = 0;
        dfs(root, -1, tree, totalWeight);
        return totalWeight;
    }

private:
    bool dfs(int node, int parent,
             unordered_map<int, vector<pair<int, int>>>& tree,
             int& totalWeight) {

        bool hasNonLeafChild = false;

        for (auto& [child, weight] : tree[node]) {
            if (child == parent) continue;

            bool childHasDescendants = dfs(child, node, tree, totalWeight);

            if (!childHasDescendants) {
                // Child is a leaf, add weight to cut it
                totalWeight += weight;
            } else {
                hasNonLeafChild = true;
            }
        }

        return hasNonLeafChild || tree[node].size() > 1;
    }
};
```

## 14. Expression Evaluation with Stack

### Problem Statement
Evaluate mathematical expressions with nested function calls like add, sub, mul, pow using a stack-based approach.

### Example
```
Input: "add(sub(-5,10),mul(5,pow(2,5)))"
Output: -15 + 160 = 145

Breakdown:
- sub(-5, 10) = -15
- pow(2, 5) = 32
- mul(5, 32) = 160
- add(-15, 160) = 145
```

### C++ Solution
```cpp
class Solution {
public:
    int evaluate(string expr) {
        stack<string> operators;
        stack<vector<int>> operands;
        string current = "";

        for (int i = 0; i < expr.size(); i++) {
            char c = expr[i];

            if (isalpha(c)) {
                current += c;
            } else if (c == '(') {
                operators.push(current);
                operands.push(vector<int>());
                current = "";
            } else if (isdigit(c) || c == '-') {
                string num;
                if (c == '-') {
                    num += c;
                    i++;
                }
                while (i < expr.size() && isdigit(expr[i])) {
                    num += expr[i];
                    i++;
                }
                i--;
                operands.top().push_back(stoi(num));
            } else if (c == ')') {
                string op = operators.top();
                operators.pop();
                vector<int> args = operands.top();
                operands.pop();

                int result = compute(op, args);

                if (!operands.empty()) {
                    operands.top().push_back(result);
                } else {
                    return result;
                }
            }
        }

        return 0;
    }

private:
    int compute(string op, vector<int>& args) {
        if (op == "add") {
            return args[0] + args[1];
        } else if (op == "sub") {
            return args[0] - args[1];
        } else if (op == "mul") {
            return args[0] * args[1];
        } else if (op == "pow") {
            return pow(args[0], args[1]);
        }
        return 0;
    }
};
```

## 15. Find Ones in Binary Array Using Query Function

### Problem Statement
Given a query function `query(L, R)` that returns 1 if subarray [L, R] contains at least one '1', find all indices with '1' in a binary array. Array size M can be very large (2^62 - 1).

**Follow-up:** query(L, R) returns count of 1s. Partition array into N/k intervals each with k ones.

### Example
```
Array: [0, 1, 1, 0, 0, 0, 1, 0, 0]
Output: [1, 2, 6]
```

### C++ Solution
```cpp
class Solution {
public:
    // Part 1: Find all indices with '1'
    vector<long long> findOnes(long long M, int N,
                               function<int(long long, long long)> query) {
        vector<long long> result;
        findOnesHelper(0, M - 1, query, result);
        return result;
    }

private:
    void findOnesHelper(long long left, long long right,
                        function<int(long long, long long)>& query,
                        vector<long long>& result) {
        if (left > right) return;

        if (query(left, right) == 0) return;

        if (left == right) {
            result.push_back(left);
            return;
        }

        long long mid = left + (right - left) / 2;
        findOnesHelper(left, mid, query, result);
        findOnesHelper(mid + 1, right, query, result);
    }

public:
    // Part 2: Partition into N/k intervals with k ones each
    vector<pair<long long, long long>> partitionWithKOnes(
        long long M, int N, int k,
        function<int(long long, long long)> query) {

        vector<pair<long long, long long>> intervals;
        long long left = 0;

        while (left < M) {
            long long right = findKthOne(left, M - 1, k, query);
            if (right == -1) break;

            intervals.push_back({left, right});
            left = right + 1;
        }

        return intervals;
    }

private:
    long long findKthOne(long long left, long long right, int k,
                         function<int(long long, long long)>& query) {
        if (left > right) return -1;

        // Binary search for position where count becomes k
        long long lo = left, hi = right, result = -1;

        while (lo <= hi) {
            long long mid = lo + (hi - lo) / 2;
            int count = query(left, mid);

            if (count >= k) {
                result = mid;
                hi = mid - 1;
            } else {
                lo = mid + 1;
            }
        }

        return result;
    }
};
```

## 16. Binary Search for Unit Tests

### Problem Statement
Use binary search to find the first failing unit test in a sequence where tests pass initially then start failing.

### Example
```
Input: tests = [pass, pass, pass, fail, fail, fail]
Output: 3 (index of first failure)
```

### C++ Solution
```cpp
class Solution {
public:
    int findFirstFailure(function<bool(int)> runTest, int n) {
        int left = 0, right = n - 1;
        int result = n;

        while (left <= right) {
            int mid = left + (right - left) / 2;

            if (!runTest(mid)) {
                // Test failed
                result = mid;
                right = mid - 1;
            } else {
                // Test passed
                left = mid + 1;
            }
        }

        return result;
    }

    // With retry logic
    int findFirstFailureWithRetry(function<bool(int)> runTest, int n, int maxRetries = 3) {
        int left = 0, right = n - 1;
        int result = n;

        while (left <= right) {
            int mid = left + (right - left) / 2;

            bool testResult = false;
            for (int retry = 0; retry < maxRetries; retry++) {
                if (runTest(mid)) {
                    testResult = true;
                    break;
                }
            }

            if (!testResult) {
                result = mid;
                right = mid - 1;
            } else {
                left = mid + 1;
            }
        }

        return result;
    }
};
```

## 18. Router Message Broadcasting

### Problem Statement
N routers are connected with distances. A source router broadcasts a message to all routers within distance D. Determine if destination router receives the message.

**Follow-up:** Message only goes to first nearest router (single hop).

### Example
```
Input:
routers = {0, 1, 2, 3}
edges = [(0,1,5), (1,2,3), (2,3,7), (0,3,20)]
src = 0, dest = 3, d = 10

Output: true (0→1→2, total distance = 8 ≤ 10)
```

### C++ Solution
```cpp
class Solution {
public:
    // Part 1: BFS within distance D
    bool canReachWithinDistance(int n, vector<vector<int>>& edges,
                                 int src, int dest, int maxDist) {
        // Build adjacency list
        unordered_map<int, vector<pair<int, int>>> graph;
        for (auto& edge : edges) {
            graph[edge[0]].push_back({edge[1], edge[2]});
            graph[edge[1]].push_back({edge[0], edge[2]});
        }

        // BFS with distance tracking
        queue<pair<int, int>> q; // {router, distance}
        unordered_set<int> visited;

        q.push({src, 0});
        visited.insert(src);

        while (!q.empty()) {
            auto [router, dist] = q.front();
            q.pop();

            if (router == dest) return true;

            for (auto [next, edgeDist] : graph[router]) {
                int newDist = dist + edgeDist;

                if (newDist <= maxDist && visited.find(next) == visited.end()) {
                    visited.insert(next);
                    q.push({next, newDist});
                }
            }
        }

        return false;
    }

    // Part 2: Single hop (only to nearest router)
    bool canReachSingleHop(int n, vector<vector<int>>& edges,
                           int src, int dest, int maxDist) {
        // Find nearest router from source
        int minDist = INT_MAX;
        int nearest = -1;

        for (auto& edge : edges) {
            if (edge[0] == src && edge[2] < minDist) {
                minDist = edge[2];
                nearest = edge[1];
            } else if (edge[1] == src && edge[2] < minDist) {
                minDist = edge[2];
                nearest = edge[0];
            }
        }

        if (nearest == -1 || minDist > maxDist) return false;
        if (nearest == dest) return true;

        // Continue from nearest router
        return canReachSingleHop(n, edges, nearest, dest, maxDist - minDist);
    }
};
```
