## 1. Detect Squares

**Problem Link:** [LeetCode 2013](https://leetcode.com/problems/detect-squares/)

### Problem Statement
Design a data structure that supports adding points to a 2D plane and counting the number of axis-aligned squares that can be formed with a new point.

### Example
```
Input:
["DetectSquares", "add", "add", "add", "count", "count", "add", "count"]
[[], [[3, 10]], [[11, 2]], [[3, 2]], [[11, 10]], [[14, 8]], [[11, 2]], [[11, 10]]]

Output:
[null, null, null, null, 1, 0, null, 2]
```

### C++ Solution
```cpp
class DetectSquares {
private:
    unordered_map<int, unordered_map<int, int>> points;
    
public:
    DetectSquares() {}
    
    void add(vector<int> point) {
        points[point[0]][point[1]]++;
    }
    
    int count(vector<int> point) {
        int x1 = point[0], y1 = point[1];
        int result = 0;
        
        // Iterate through all points with same x-coordinate
        if (points.find(x1) == points.end()) return 0;
        
        for (auto& [y3, cnt1] : points[x1]) {
            if (y3 == y1) continue;
            int sideLen = abs(y3 - y1);
            
            // Check for squares on both sides
            for (int dx : {-sideLen, sideLen}) {
                int x2 = x1 + dx;
                if (points.find(x2) != points.end()) {
                    result += cnt1 * points[x2][y1] * points[x2][y3];
                }
            }
        }
        
        return result;
    }
};
```

## 2. Tetris Game Implementation

### Problem Statement
Implement a simple Tetris game. Design the core mechanics including:
- Grid representation
- Piece dropping
- Row clearing
- Collision detection

### Example
```
Input: 
Grid: 10x20
Piece: I-piece (vertical line of 4 blocks)
Position: column 5

Output:
Update grid with piece placed at bottom or on top of existing blocks
Clear any complete rows
Return updated grid state
```

### C++ Solution
```cpp
class Tetris {
private:
    vector<vector<int>> grid;
    int rows, cols;
    
public:
    Tetris(int r, int c) : rows(r), cols(c) {
        grid.resize(rows, vector<int>(cols, 0));
    }
    
    bool canPlace(vector<pair<int,int>>& piece, int row, int col) {
        for (auto& [dr, dc] : piece) {
            int newRow = row + dr, newCol = col + dc;
            if (newRow < 0 || newRow >= rows || newCol < 0 || newCol >= cols)
                return false;
            if (grid[newRow][newCol] != 0)
                return false;
        }
        return true;
    }
    
    void placePiece(vector<pair<int,int>>& piece, int col) {
        int row = 0;
        // Drop piece until it can't move down
        while (canPlace(piece, row + 1, col)) {
            row++;
        }
        
        // Place piece
        for (auto& [dr, dc] : piece) {
            grid[row + dr][col + dc] = 1;
        }
        
        clearRows();
    }
    
    void clearRows() {
        for (int i = rows - 1; i >= 0; i--) {
            bool full = true;
            for (int j = 0; j < cols; j++) {
                if (grid[i][j] == 0) {
                    full = false;
                    break;
                }
            }
            
            if (full) {
                grid.erase(grid.begin() + i);
                grid.insert(grid.begin(), vector<int>(cols, 0));
                i++; // Check same row again
            }
        }
    }
    
    vector<vector<int>> getGrid() {
        return grid;
    }
};
```

## 3. Partition Array with Negative Numbers

### Problem Statement
Given an array of integers, find the number of ways to partition the array into segments such that each segment contains at least 2 negative numbers.

### Example
```
Input: nums = [-1, 2, -3, -4, 5, -6, -7]
Output: 3

Explanation:
Possible partitions:
1. [-1, 2, -3, -4] | [5, -6, -7]
2. [-1, 2, -3, -4, 5, -6, -7]
3. [-1, 2, -3] | [-4, 5, -6, -7]
```

### C++ Solution
```cpp
class Solution {
public:
    int countPartitions(vector<int>& nums) {
        int n = nums.size();
        vector<int> dp(n + 1, 0);
        dp[0] = 1;
        
        for (int i = 1; i <= n; i++) {
            int negCount = 0;
            
            // Try all possible last segments ending at i
            for (int j = i - 1; j >= 0; j--) {
                if (nums[j] < 0) negCount++;
                
                // If segment [j, i-1] has at least 2 negatives
                if (negCount >= 2) {
                    dp[i] += dp[j];
                }
            }
        }
        
        return dp[n];
    }
};

// Optimized O(N) approach
class SolutionOptimized {
public:
    int countPartitions(vector<int>& nums) {
        int n = nums.size();
        vector<int> dp(n + 1, 0);
        dp[0] = 1;
        
        int sum = 0;
        int negCount = 0;
        int lastValid = -1;
        
        for (int i = 0; i < n; i++) {
            if (nums[i] < 0) negCount++;
            
            if (negCount >= 2) {
                // Find first position where we have >= 2 negatives
                if (lastValid == -1) {
                    int tempNeg = 0;
                    for (int j = i; j >= 0; j--) {
                        if (nums[j] < 0) tempNeg++;
                        if (tempNeg >= 2) {
                            lastValid = j;
                            break;
                        }
                    }
                }
                
                if (lastValid != -1) {
                    dp[i + 1] = sum + dp[lastValid];
                }
            }
            
            sum += dp[i];
        }
        
        return dp[n];
    }
};
```

## 4. Prefix Search in Dictionary

### Problem Statement
Given a dictionary of words, design a data structure that efficiently finds how many words have a given key as a prefix.

### Example
```
Input:
Dictionary: ["apple", "app", "apricot", "banana", "band"]
Queries: ["ap", "ban", "cat"]

Output: [3, 2, 0]

Explanation:
- "ap" is prefix of: apple, app, apricot (3 words)
- "ban" is prefix of: banana, band (2 words)
- "cat" is prefix of: none (0 words)
```

### C++ Solution
```cpp
class TrieNode {
public:
    unordered_map<char, TrieNode*> children;
    int count;
    
    TrieNode() : count(0) {}
};

class PrefixDictionary {
private:
    TrieNode* root;
    
public:
    PrefixDictionary() {
        root = new TrieNode();
    }
    
    void insert(string word) {
        TrieNode* node = root;
        for (char c : word) {
            if (node->children.find(c) == node->children.end()) {
                node->children[c] = new TrieNode();
            }
            node = node->children[c];
            node->count++;
        }
    }
    
    int countPrefix(string prefix) {
        TrieNode* node = root;
        for (char c : prefix) {
            if (node->children.find(c) == node->children.end()) {
                return 0;
            }
            node = node->children[c];
        }
        return node->count;
    }
    
    vector<int> searchMultiple(vector<string>& queries) {
        vector<int> result;
        for (string& query : queries) {
            result.push_back(countPrefix(query));
        }
        return result;
    }
};
```

## 5. Five Words Without Duplicate Characters

### Problem Statement
Given a list of strings (each exactly 5 characters long), find all combinations of 5 words that can form a sentence where no character appears more than once across all 5 words.

### Example
```
Input: words = ["werty", "uiopa", "sdfgh", "jklzx", "cvbnm", "aasdg", "sdfjk", "lzxcv", "bnmgh"]

Output: 
[
  ["werty", "uiopa", "sdfgh", "jklzx", "cvbnm"],
  ["werty", "uiopa", "sdfjk", "lzxcv", "bnmgh"]
]
```

### C++ Solution
```cpp
class Solution {
private:
    vector<vector<string>> result;
    
    struct WordMask {
        string word;
        int mask;
    };
    
public:
    vector<vector<string>> findValidCombinations(vector<string>& words) {
        vector<WordMask> validWords;
        
        // Filter words without duplicate characters
        for (const string& word : words) {
            int mask = 0;
            bool isValid = true;
            
            for (char c : word) {
                int bit = c - 'a';
                if (mask & (1 << bit)) {
                    isValid = false;
                    break;
                }
                mask |= (1 << bit);
            }
            
            if (isValid) {
                validWords.push_back({word, mask});
            }
        }
        
        vector<string> current;
        backtrack(validWords, 0, 0, current);
        return result;
    }
    
private:
    void backtrack(vector<WordMask>& words, int idx, int mask, vector<string>& current) {
        if (current.size() == 5) {
            result.push_back(current);
            return;
        }
        
        for (int i = idx; i < words.size(); i++) {
            if ((mask & words[i].mask) == 0) {
                current.push_back(words[i].word);
                backtrack(words, i + 1, mask | words[i].mask, current);
                current.pop_back();
            }
        }
    }
};
```

## 6. Place Galaxy in Universe (N-Dimensional Grid)

### Problem Statement
Design a system to place galaxies in an n-dimensional grid universe. Each galaxy occupies a hypercube of a given size. Implement functions to add galaxies and check if a position is valid (no overlap with existing galaxies).

### Example
```
Input (2D case):
addGalaxy(0, 0, 3)  // Galaxy at (0,0) with size 3
canPlace(2, 2, 2)   // Can we place a galaxy of size 2 at (2,2)?

Output: false (overlaps with existing galaxy)
```

### C++ Solution
```cpp
class UniverseGrid {
private:
    struct Galaxy {
        vector<int> position;
        int size;
    };
    
    vector<Galaxy> galaxies;
    int dimensions;
    
public:
    UniverseGrid(int dim) : dimensions(dim) {}
    
    bool hasOverlap(const vector<int>& pos1, int size1, 
                    const vector<int>& pos2, int size2) {
        for (int i = 0; i < dimensions; i++) {
            int end1 = pos1[i] + size1;
            int end2 = pos2[i] + size2;
            
            // No overlap in this dimension
            if (pos1[i] >= end2 || pos2[i] >= end1) {
                return false;
            }
        }
        return true;
    }
    
    bool canPlace(vector<int>& position, int size) {
        for (const Galaxy& g : galaxies) {
            if (hasOverlap(position, size, g.position, g.size)) {
                return false;
            }
        }
        return true;
    }
    
    bool addGalaxy(vector<int>& position, int size) {
        if (!canPlace(position, size)) {
            return false;
        }
        galaxies.push_back({position, size});
        return true;
    }
    
    vector<Galaxy> getGalaxies() {
        return galaxies;
    }
};

// Example usage for 2D case
class Solution2D {
public:
    bool canPlaceGalaxy(vector<vector<int>>& existingGalaxies, 
                        int x, int y, int size) {
        for (auto& galaxy : existingGalaxies) {
            int gx = galaxy[0], gy = galaxy[1], gsize = galaxy[2];
            
            // Check if ranges overlap in both dimensions
            bool xOverlap = !(x >= gx + gsize || gx >= x + size);
            bool yOverlap = !(y >= gy + gsize || gy >= y + size);
            
            if (xOverlap && yOverlap) {
                return false;
            }
        }
        return true;
    }
};
```

## 7. Restaurant Waitlist Management

### Problem Statement
Design a waitlist system for a restaurant where:
- Groups can be added to the waitlist with their party size
- Groups can leave the waitlist at any time (by unique ID)
- When a table becomes available, select the first (oldest) group that exactly fits the table size

**Requirements:**
- `addGroup(id, size)`: O(1)
- `removeGroup(id)`: O(1)
- `selectGroup(tableSize)`: O(log N) or better

### Example
```
Input:
addGroup("Bob", 4)
addGroup("Alice", 2)
addGroup("Charlie", 4)
selectGroup(4)  // Returns "Bob"
removeGroup("Alice")
selectGroup(4)  // Returns "Charlie"
```

### C++ Solution
```cpp
class RestaurantWaitlist {
private:
    struct Group {
        string id;
        int size;
        int timestamp;
        
        Group(string i, int s, int t) : id(i), size(s), timestamp(t) {}
    };
    
    // Map from group ID to iterator in the list
    unordered_map<string, list<Group>::iterator> groupMap;
    
    // Map from party size to ordered list of groups (FIFO)
    unordered_map<int, list<Group>> sizeToGroups;
    
    int currentTime;
    
public:
    RestaurantWaitlist() : currentTime(0) {}
    
    void addGroup(string id, int size) {
        sizeToGroups[size].push_back(Group(id, size, currentTime++));
        auto it = --sizeToGroups[size].end();
        groupMap[id] = it;
    }
    
    void removeGroup(string id) {
        if (groupMap.find(id) == groupMap.end()) return;
        
        auto it = groupMap[id];
        int size = it->size;
        
        sizeToGroups[size].erase(it);
        if (sizeToGroups[size].empty()) {
            sizeToGroups.erase(size);
        }
        
        groupMap.erase(id);
    }
    
    string selectGroup(int tableSize) {
        if (sizeToGroups.find(tableSize) == sizeToGroups.end() || 
            sizeToGroups[tableSize].empty()) {
            return "";
        }
        
        Group selected = sizeToGroups[tableSize].front();
        sizeToGroups[tableSize].pop_front();
        
        if (sizeToGroups[tableSize].empty()) {
            sizeToGroups.erase(tableSize);
        }
        
        groupMap.erase(selected.id);
        return selected.id;
    }
};

// Alternative solution using ordered map for better selectGroup
class RestaurantWaitlistOptimized {
private:
    struct Group {
        string id;
        int size;
        int order;
    };
    
    unordered_map<string, Group> groups;
    map<int, queue<string>> sizeToQueue; // size -> queue of IDs
    int orderCounter;
    
public:
    RestaurantWaitlistOptimized() : orderCounter(0) {}
    
    void addGroup(string id, int size) {
        groups[id] = {id, size, orderCounter++};
        sizeToQueue[size].push(id);
    }
    
    void removeGroup(string id) {
        if (groups.find(id) == groups.end()) return;
        groups.erase(id);
        // Note: ID remains in queue but will be skipped in selectGroup
    }
    
    string selectGroup(int tableSize) {
        if (sizeToQueue.find(tableSize) == sizeToQueue.end()) {
            return "";
        }
        
        auto& q = sizeToQueue[tableSize];
        while (!q.empty()) {
            string id = q.front();
            q.pop();
            
            if (groups.find(id) != groups.end()) {
                groups.erase(id);
                return id;
            }
        }
        
        return "";
    }
};
```

## 8. Build Tree from Dictionary with Parent References

### Problem Statement
Given a list of dictionaries where each dictionary contains key-value pairs and a parent reference, build a tree structure with references to children nodes.

### Example
```
Input:
[
  {id: 1, name: "root", parent: null},
  {id: 2, name: "child1", parent: 1},
  {id: 3, name: "child2", parent: 1},
  {id: 4, name: "grandchild", parent: 2}
]

Output: Tree structure where node 1 has children [2, 3] and node 2 has child [4]
```

### C++ Solution
```cpp
class TreeNode {
public:
    int id;
    string name;
    vector<TreeNode*> children;
    
    TreeNode(int i, string n) : id(i), name(n) {}
};

class Solution {
public:
    TreeNode* buildTree(vector<map<string, string>>& data) {
        unordered_map<int, TreeNode*> nodes;
        TreeNode* root = nullptr;
        
        // First pass: create all nodes
        for (auto& dict : data) {
            int id = stoi(dict["id"]);
            string name = dict["name"];
            nodes[id] = new TreeNode(id, name);
        }
        
        // Second pass: establish parent-child relationships
        for (auto& dict : data) {
            int id = stoi(dict["id"]);
            
            if (dict["parent"] == "null" || dict.find("parent") == dict.end()) {
                root = nodes[id];
            } else {
                int parentId = stoi(dict["parent"]);
                nodes[parentId]->children.push_back(nodes[id]);
            }
        }
        
        return root;
    }
    
    // Alternative: if input is already structured
    TreeNode* buildTreeFromVector(vector<tuple<int, string, int>>& data) {
        // data: (id, name, parentId), where parentId = -1 means root
        unordered_map<int, TreeNode*> nodes;
        TreeNode* root = nullptr;
        
        for (auto& [id, name, parentId] : data) {
            if (nodes.find(id) == nodes.end()) {
                nodes[id] = new TreeNode(id, name);
            }
            
            if (parentId == -1) {
                root = nodes[id];
            } else {
                if (nodes.find(parentId) == nodes.end()) {
                    nodes[parentId] = new TreeNode(parentId, "");
                }
                nodes[parentId]->children.push_back(nodes[id]);
            }
        }
        
        return root;
    }
};
```

## 9. Maximum Score Stone Jump

### Problem Statement
Given a sequence of stones with non-negative integers, calculate the maximum possible score from the beginning to the end. You can jump any number of positions from each position. The score for each jump is calculated as: `destination_stone_value × number_of_positions_jumped`.

The first stone's value doesn't contribute to the score (you start there).

### Example
```
Input: stones = [1, 2, 3, 4, 5]
Output: 20
Explanation: Jump from index 0 to index 4 → 5 × 4 = 20

Input: stones = [5, 4, 3, 2, 1]
Output: 10
Explanation: Jump one position at a time → 4 + 3 + 2 + 1 = 10

Input: stones = [2, 4, 6, 8, 10]
Output: 40
Explanation: Jump to last → 10 × 4 = 40
```

### C++ Solution
```cpp
class Solution {
public:
    // O(n^2) Dynamic Programming Solution
    int maxScore(vector<int>& stones) {
        int n = stones.size();
        if (n <= 1) return 0;
        
        vector<long long> dp(n, 0);
        
        for (int i = 1; i < n; i++) {
            for (int j = 0; j < i; j++) {
                long long score = dp[j] + (long long)stones[i] * (i - j);
                dp[i] = max(dp[i], score);
            }
        }
        
        return dp[n - 1];
    }
    
    // Optimized O(n log n) solution using monotonic deque/stack
    int maxScoreOptimized(vector<int>& stones) {
        int n = stones.size();
        if (n <= 1) return 0;
        
        vector<long long> dp(n, 0);
        deque<int> dq; // monotonic deque to maintain candidates
        
        for (int i = 1; i < n; i++) {
            // For each position i, we want to maximize:
            // dp[j] + stones[i] * (i - j) = dp[j] - stones[i] * j + stones[i] * i
            // = (dp[j] - stones[i] * j) + stones[i] * i
            
            long long maxVal = LLONG_MIN;
            
            // Remove elements that are no longer optimal
            while (!dq.empty()) {
                int j = dq.back();
                long long val1 = dp[j] - (long long)stones[i] * j;
                long long val2 = dp[dq.back()] - (long long)stones[i] * dq.back();
                
                // This is a simplified version - actual optimization needs
                // convex hull trick or segment tree for true O(n log n)
                if (val1 >= maxVal) {
                    maxVal = val1;
                }
                break;
            }
            
            // Compute best score for current position
            for (int j = 0; j < i; j++) {
                long long score = dp[j] + (long long)stones[i] * (i - j);
                dp[i] = max(dp[i], score);
            }
        }
        
        return dp[n - 1];
    }
    
    // Alternative approach with better complexity using slope optimization
    int maxScoreSlopeOptimization(vector<int>& stones) {
        int n = stones.size();
        if (n <= 1) return 0;
        
        vector<long long> dp(n, 0);
        
        // Use convex hull trick for optimization
        // For each i, we want max of: dp[j] + stones[i] * (i - j)
        // Rewrite as: (dp[j] - j * stones[i]) + i * stones[i]
        
        for (int i = 1; i < n; i++) {
            long long maxScore = 0;
            for (int j = 0; j < i; j++) {
                long long score = dp[j] + (long long)stones[i] * (i - j);
                maxScore = max(maxScore, score);
            }
            dp[i] = maxScore;
        }
        
        return dp[n - 1];
    }
};
```

### Car Fleet (LeetCode 853)
**Problem Link:** [LeetCode 853](https://leetcode.com/problems/car-fleet/)

### C++ Solution
```cpp
class Solution {
public:
    int carFleet(int target, vector<int>& position, vector<int>& speed) {
        int n = position.size();
        vector<pair<int, double>> cars;
        
        // Calculate time to reach target for each car
        for (int i = 0; i < n; i++) {
            double time = (double)(target - position[i]) / speed[i];
            cars.push_back({position[i], time});
        }
        
        // Sort by position (descending)
        sort(cars.begin(), cars.end(), greater<pair<int, double>>());
        
        int fleets = 0;
        double prevTime = 0;
        
        for (auto& [pos, time] : cars) {
            if (time > prevTime) {
                fleets++;
                prevTime = time;
            }
        }
        
        return fleets;
    }
};
```
