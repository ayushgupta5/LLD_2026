## 1. Check if Word Can Be Placed in Crossword

**Problem Link:** [LeetCode 2018](https://leetcode.com/problems/check-if-word-can-be-placed-in-crossword/)

### Problem Statement
Given an `m x n` board where each cell is a letter, a blocked cell ('#'), or an empty cell (' '), determine if a word can be placed on the board either horizontally or vertically.

### Example
```
Input: board = [["#", " ", "#"], [" ", " ", "#"], ["#", "c", " "]], word = "abc"
Output: true

Input: board = [[" ", "#", "a"], [" ", "#", "c"], [" ", "#", "a"]], word = "abc"
Output: false
```

### C++ Solution
```cpp
class Solution {
public:
    bool placeWordInCrossword(vector<vector<char>>& board, string word) {
        int m = board.size(), n = board[0].size();
        int len = word.size();

        // Check horizontal placements
        for (int i = 0; i < m; i++) {
            for (int j = 0; j <= n - len; j++) {
                if (canPlaceHorizontal(board, word, i, j, len)) {
                    return true;
                }
            }
        }

        // Check vertical placements
        for (int i = 0; i <= m - len; i++) {
            for (int j = 0; j < n; j++) {
                if (canPlaceVertical(board, word, i, j, len)) {
                    return true;
                }
            }
        }

        return false;
    }

private:
    bool canPlaceHorizontal(vector<vector<char>>& board, string& word, int row, int col, int len) {
        // Check boundaries
        if (col > 0 && board[row][col - 1] != '#') return false;
        if (col + len < board[0].size() && board[row][col + len] != '#') return false;

        // Check forward
        if (matches(board, word, row, col, len, true, false)) return true;
        // Check backward
        if (matches(board, word, row, col, len, true, true)) return true;

        return false;
    }

    bool canPlaceVertical(vector<vector<char>>& board, string& word, int row, int col, int len) {
        // Check boundaries
        if (row > 0 && board[row - 1][col] != '#') return false;
        if (row + len < board.size() && board[row + len][col] != '#') return false;

        // Check forward
        if (matches(board, word, row, col, len, false, false)) return true;
        // Check backward
        if (matches(board, word, row, col, len, false, true)) return true;

        return false;
    }

    bool matches(vector<vector<char>>& board, string& word, int row, int col,
                 int len, bool horizontal, bool reverse) {
        for (int i = 0; i < len; i++) {
            char c = reverse ? word[len - 1 - i] : word[i];
            char cell = horizontal ? board[row][col + i] : board[row + i][col];

            if (cell != ' ' && cell != c) {
                return false;
            }
        }
        return true;
    }
};

// Follow-up: Filter list of words that can be placed
class SolutionWithFilter {
public:
    vector<string> filterPlaceableWords(vector<vector<char>>& board, vector<string>& words) {
        vector<string> result;
        Solution solver;

        for (const string& word : words) {
            if (solver.placeWordInCrossword(board, word)) {
                result.push_back(word);
            }
        }

        return result;
    }
};
```

## 2. Swipe Keyboard Word Suggestion

### Problem Statement
Implement a word suggestion engine for swipe keyboards on phones. Given a user's swipe path (sequence of characters) and a dictionary of words, find the best matching word. The swipe path contains all characters that fall under the user's finger movement, which may include extra characters.

### Example
```
Input:
swipePath = "qwertyuiujhgcvghbnjk"
dictionary = ["aardvark", "angel", "dog", "quick", "quickly", "quickening", "TikTok", "zebra"]

Output: "quick"

Explanation: The subsequence "q-u-i-c-k" exists in the swipe path
```

### C++ Solution
```cpp
class TrieNode {
public:
    unordered_map<char, TrieNode*> children;
    string word;

    TrieNode() : word("") {}
};

class SwipeKeyboard {
private:
    TrieNode* root;

public:
    SwipeKeyboard(vector<string>& dictionary) {
        root = new TrieNode();
        buildTrie(dictionary);
    }

    void buildTrie(vector<string>& dictionary) {
        for (string& word : dictionary) {
            TrieNode* node = root;
            for (char c : word) {
                char lower = tolower(c);
                if (node->children.find(lower) == node->children.end()) {
                    node->children[lower] = new TrieNode();
                }
                node = node->children[lower];
            }
            node->word = word;
        }
    }

    string suggestWord(string swipePath) {
        vector<string> matches;
        dfs(root, swipePath, 0, matches);

        // Return shortest match (most likely intended word)
        if (matches.empty()) return "";

        string best = matches[0];
        for (const string& match : matches) {
            if (match.length() < best.length()) {
                best = match;
            }
        }
        return best;
    }

private:
    void dfs(TrieNode* node, string& path, int idx, vector<string>& matches) {
        if (!node->word.empty()) {
            matches.push_back(node->word);
        }

        if (idx >= path.size()) return;

        // Try to match current character
        char c = tolower(path[idx]);
        if (node->children.find(c) != node->children.end()) {
            dfs(node->children[c], path, idx + 1, matches);
        }

        // Skip current character (allows for mistakes)
        dfs(node, path, idx + 1, matches);
    }
};

// Version with error tolerance
class SwipeKeyboardWithErrors {
private:
    TrieNode* root;

public:
    SwipeKeyboardWithErrors(vector<string>& dictionary) {
        root = new TrieNode();
        for (string& word : dictionary) {
            TrieNode* node = root;
            for (char c : word) {
                char lower = tolower(c);
                if (node->children.find(lower) == node->children.end()) {
                    node->children[lower] = new TrieNode();
                }
                node = node->children[lower];
            }
            node->word = word;
        }
    }

    string suggestWordWithErrors(string swipePath, int maxErrors = 2) {
        vector<pair<string, int>> matches;
        dfsWithErrors(root, swipePath, 0, 0, maxErrors, matches);

        if (matches.empty()) return "";

        // Sort by errors (fewer is better), then by length
        sort(matches.begin(), matches.end(),
             [](const pair<string, int>& a, const pair<string, int>& b) {
                 if (a.second != b.second) return a.second < b.second;
                 return a.first.length() < b.first.length();
             });

        return matches[0].first;
    }

private:
    void dfsWithErrors(TrieNode* node, string& path, int idx, int errors,
                       int maxErrors, vector<pair<string, int>>& matches) {
        if (!node->word.empty()) {
            matches.push_back({node->word, errors});
        }

        if (idx >= path.size()) return;
        if (errors > maxErrors) return;

        char c = tolower(path[idx]);

        // Match current character correctly
        if (node->children.find(c) != node->children.end()) {
            dfsWithErrors(node->children[c], path, idx + 1, errors, maxErrors, matches);
        }

        // Consider current character as an error (skip in path)
        dfsWithErrors(node, path, idx + 1, errors + 1, maxErrors, matches);

        // Consider missing a character in word (advance in trie without advancing path)
        for (auto& [ch, child] : node->children) {
            dfsWithErrors(child, path, idx, errors + 1, maxErrors, matches);
        }
    }
};
```

## 3. Weighted Random Key Generator

**Problem Link:** [LeetCode 528 - Random Pick with Weight](https://leetcode.com/problems/random-pick-with-weight/)

### Problem Statement
Design a data structure that picks a random key based on its weight. Higher weight keys should be picked more frequently.

### Example
```
Input:
keys = ["A", "B", "C"]
weights = [1, 2, 3]

pickRandom() might return:
- "A" with probability 1/6
- "B" with probability 2/6
- "C" with probability 3/6
```

### C++ Solution
```cpp
class WeightedRandomPicker {
private:
    vector<string> keys;
    vector<double> prefixSum;
    double totalWeight;

public:
    WeightedRandomPicker(vector<string>& k, vector<double>& weights) {
        if (k.size() != weights.size()) {
            throw invalid_argument("Keys and weights must have same size");
        }

        keys = k;
        totalWeight = 0;

        for (double w : weights) {
            if (w < 0) {
                throw invalid_argument("Weight cannot be negative");
            }
            if (w == 0) {
                throw invalid_argument("Weight cannot be zero");
            }
            totalWeight += w;
            prefixSum.push_back(totalWeight);
        }
    }

    string pickRandom() {
        double random = ((double) rand() / RAND_MAX) * totalWeight;

        // Binary search for the index
        int left = 0, right = prefixSum.size() - 1;

        while (left < right) {
            int mid = left + (right - left) / 2;
            if (prefixSum[mid] < random) {
                left = mid + 1;
            } else {
                right = mid;
            }
        }

        return keys[left];
    }
};

// Integer weight version
class WeightedRandomPickerInt {
private:
    vector<string> keys;
    vector<int> prefixSum;
    int totalWeight;

public:
    WeightedRandomPickerInt(vector<string>& k, vector<int>& weights) : totalWeight(0) {
        keys = k;

        for (int w : weights) {
            totalWeight += w;
            prefixSum.push_back(totalWeight);
        }
    }

    string pickRandom() {
        int random = rand() % totalWeight;

        int left = 0, right = prefixSum.size() - 1;
        while (left < right) {
            int mid = left + (right - left) / 2;
            if (prefixSum[mid] <= random) {
                left = mid + 1;
            } else {
                right = mid;
            }
        }

        return keys[left];
    }
};
```

## 4. Stream Triplet with Distance Constraint

### Problem Statement
Given a stream of unique floating-point numbers, find and remove any triplet (a, b, c) from the stream where `|a-b|`, `|b-c|`, and `|a-c|` are all ≤ d. Return the triplet if found, otherwise add the number to the stream.

### Example
```
Input: stream = [1.0, 2.5, 3.0, 1.8], d = 1.5
Output:
- After 1.0: store
- After 2.5: store [1.0, 2.5]
- After 3.0: store [1.0, 2.5, 3.0]
- After 1.8: return [1.0, 1.8, 2.5] (all differences ≤ 1.5)
```

### C++ Solution
```cpp
class TripletStream {
private:
    set<double> stream;
    double maxDist;

public:
    TripletStream(double d) : maxDist(d) {}

    vector<double> processNumber(double num) {
        // Check for valid triplets with current number
        auto it = stream.lower_bound(num);

        // Get 2 elements smaller than num
        vector<double> candidates;
        auto temp = it;

        for (int i = 0; i < 2 && temp != stream.begin(); i++) {
            --temp;
            candidates.push_back(*temp);
        }

        // Get 2 elements greater than num
        temp = it;
        for (int i = 0; i < 2 && temp != stream.end(); i++) {
            candidates.push_back(*temp);
            ++temp;
        }

        // Check all possible triplets
        candidates.push_back(num);
        sort(candidates.begin(), candidates.end());

        for (int i = 0; i + 2 < candidates.size(); i++) {
            double a = candidates[i];
            double b = candidates[i + 1];
            double c = candidates[i + 2];

            if (abs(a - b) <= maxDist &&
                abs(b - c) <= maxDist &&
                abs(a - c) <= maxDist) {

                // Remove from stream
                if (a != num) stream.erase(a);
                if (b != num) stream.erase(b);
                if (c != num) stream.erase(c);

                return {a, b, c};
            }
        }

        // No valid triplet found, insert number
        stream.insert(num);
        return {};
    }

    set<double> getStream() {
        return stream;
    }
};
```

## 5. Find Array Partitions

**Problem Link:** [LeetCode Discussion](https://leetcode.com/discuss/interview-question/4574669/Google-or-Onsite-or-Find-partitions/)

### Problem Statement
Given an array, find the number of ways to partition it into subarrays where each subarray satisfies certain conditions (problem-specific constraints apply).

### Example
```
Input: arr = [1, 2, 3, 4], condition: sum is even
Output: 3

Partitions:
1. [1,2,3,4]
2. [1],[2,3,4] (invalid - sum of [2,3,4]=9 is odd)
3. [1,2][3,4] (invalid)
...
```

### C++ Solution
```cpp
class Solution {
public:
    // Generic partition counting with memoization
    int countPartitions(vector<int>& arr, function<bool(int, int)> isValid) {
        int n = arr.size();
        vector<int> dp(n + 1, 0);
        dp[0] = 1; // Base case: empty array has 1 way

        for (int i = 1; i <= n; i++) {
            for (int j = 0; j < i; j++) {
                // Check if subarray [j, i-1] is valid
                if (isValid(j, i - 1)) {
                    dp[i] += dp[j];
                }
            }
        }

        return dp[n];
    }

    // Example: partitions where each segment sum is even
    int countEvenSumPartitions(vector<int>& arr) {
        int n = arr.size();
        vector<int> prefixSum(n + 1, 0);

        for (int i = 0; i < n; i++) {
            prefixSum[i + 1] = prefixSum[i] + arr[i];
        }

        auto isEvenSum = [&](int start, int end) {
            int sum = prefixSum[end + 1] - prefixSum[start];
            return sum % 2 == 0;
        };

        return countPartitions(arr, isEvenSum);
    }
};

// Optimized version with specific constraint
class SolutionOptimized {
public:
    int countValidPartitions(vector<int>& arr) {

        vector<long long> dp(n + 1, 0);
        dp[0] = 1;

        for (int i = 1; i <= n; i++) {
            // Process all valid partitions ending at i
            for (int j = i - 1; j >= 0; j--) {
                // Check if segment [j, i-1] is valid
                // Add your specific validation logic here
                dp[i] += dp[j];
            }
        }

        return dp[n];
    }
};
```

## 6. Valid Signal Detection

### Problem Statement
Given two arrays A and B of size n representing signal strengths from two antennas, find the number of valid signals. A signal is valid if it appears in both arrays at positions with distance ≤ D.

### Example
```
Input:
A = [1, 3, 4, 3, 4, 5, 6]
B = [4, 1, 8, 7, 6, 3, 2]
D = 2

Output: 3

Explanation:
- A[0]=1 matches B[1]=1 (distance = 1 ≤ 2) ✓
- A[2]=4 matches B[0]=4 (distance = 2 ≤ 2) ✓
- A[3]=3 matches B[5]=3 (distance = 2 ≤ 2) ✓
```

### C++ Solution
```cpp
class Solution {
public:
    int countValidSignals(vector<int>& A, vector<int>& B, int D) {
        int n = A.size();
        int result = 0;

        unordered_map<int, int> window;

        // Initialize window with first 2D+1 elements (or min with n)
        int windowSize = min(n, 2 * D + 1);
        for (int i = 0; i < windowSize; i++) {
            window[B[i]]++;
        }

        // Process first element
        if (window.find(A[0]) != window.end() && window[A[0]] > 0) {
            result++;
            window[A[0]]--;
        }

        // Slide the window
        for (int i = 1; i < n; i++) {
            // Add new element to the right of window
            int rightIdx = i + D;
            if (rightIdx < n) {
                window[B[rightIdx]]++;
            }

            // Remove element from the left of window
            int leftIdx = i - D - 1;
            if (leftIdx >= 0) {
                window[B[leftIdx]]--;
                if (window[B[leftIdx]] == 0) {
                    window.erase(B[leftIdx]);
                }
            }

            // Check if current A[i] is in window
            if (window.find(A[i]) != window.end() && window[A[i]] > 0) {
                result++;
                window[A[i]]--;
            }
        }

        return result;
    }
};
```

## 7. Logger Rate Limiter with Window

**Problem Link:** [LeetCode 359](https://leetcode.com/problems/logger-rate-limiter/)

### Problem Statement
Design a logger system that receives a stream of messages along with timestamps. A message should only be printed if it appears exactly once in a time window [t-10, t+10].

### Example
```
Input:
["Logger","shouldPrintMessage","shouldPrintMessage","shouldPrintMessage"]
[[],["foo",1],["foo",8],["bar",10]]

Output: [null,true,false,true]

Explanation:
- "foo" at t=1: print (first occurrence)
- "foo" at t=8: don't print (within window [1-10, 1+10])
- "bar" at t=10: print (first occurrence)
```

### C++ Solution
```cpp
class Logger {
private:
    map<int, string> timeToMessage;
    unordered_map<string, int> messageToTime;
    unordered_set<string> invalidMessages;

public:
    Logger() {}

    bool shouldPrintMessage(int timestamp, string message) {
        // Clean up old messages
        cleanOldMessages(timestamp);

        // Check if message already seen
        if (messageToTime.find(message) != messageToTime.end()) {
            int prevTime = messageToTime[message];

            // If within window, mark as invalid
            if (abs(timestamp - prevTime) <= 10) {
                invalidMessages.insert(message);
                timeToMessage.erase(prevTime);
                messageToTime.erase(message);
                return false;
            }
        }

        // Check if message was previously invalidated
        if (invalidMessages.find(message) != invalidMessages.end()) {
            return false;
        }

        // Valid message
        timeToMessage[timestamp] = message;
        messageToTime[message] = timestamp;
        return true;
    }

private:
    void cleanOldMessages(int currentTime) {
        auto it = timeToMessage.begin();
        while (it != timeToMessage.end() && it->first < currentTime - 10) {
            messageToTime.erase(it->second);
            it = timeToMessage.erase(it);
        }
    }
};

// Standard Logger Rate Limiter (10 second gap)
class LoggerSimple {
private:
    unordered_map<string, int> messageTime;

public:
    LoggerSimple() {}

    bool shouldPrintMessage(int timestamp, string message) {
        if (messageTime.find(message) == messageTime.end() ||
            timestamp - messageTime[message] >= 10) {
            messageTime[message] = timestamp;
            return true;
        }
        return false;
    }
};
```

## 8. Next Word Frequency

### Problem Statement
Given a list of word sequences, build a data structure that returns the most frequent next word for any given word in O(1) time.

### Example
```
Input:
sequences = [
  ["a", "she", "he", "am"],
  ["a", "he", "she", "you", "are"],
  ["a", "he", "she", "he", "are"]
]

Queries:
- "a" → "he" (appears 2 times after "a")
- "she" → "he" (appears 2 times after "she")
- "am" → "" (no next word)
```

### C++ Solution
```cpp
class NextWordFrequency {
private:
    unordered_map<string, unordered_map<string, int>> nextWordCount;
    unordered_map<string, string> mostFrequentNext;

public:
    void addSequences(vector<vector<string>>& sequences) {
        // Build frequency map
        for (auto& seq : sequences) {
            for (int i = 0; i < seq.size() - 1; i++) {
                nextWordCount[seq[i]][seq[i + 1]]++;
            }
        }

        // Preprocess to find most frequent next word for each word
        for (auto& [word, nextWords] : nextWordCount) {
            string maxWord = "";
            int maxCount = 0;

            for (auto& [nextWord, count] : nextWords) {
                if (count > maxCount) {
                    maxCount = count;
                    maxWord = nextWord;
                }
            }

            mostFrequentNext[word] = maxWord;
        }
    }

    string getMostFrequentNext(string word) {
        if (mostFrequentNext.find(word) == mostFrequentNext.end()) {
            return "";
        }
        return mostFrequentNext[word];
    }

    // Alternative: get all next words with their frequencies
    vector<pair<string, int>> getAllNextWords(string word) {
        vector<pair<string, int>> result;

        if (nextWordCount.find(word) != nextWordCount.end()) {
            for (auto& [nextWord, count] : nextWordCount[word]) {
                result.push_back({nextWord, count});
            }

            sort(result.begin(), result.end(),
                 [](const pair<string, int>& a, const pair<string, int>& b) {
                     return a.second > b.second;
                 });
        }

        return result;
    }
};
```

## 9. Chess Tournament Definite Rankings

### Problem Statement
Given a chess tournament with N players and M match results, determine which players' ranks can be definitively determined.

### Example
```
Input:
matches = [
  ["p1", "p2", "p1"],  // p1 won
  ["p2", "p5", "p2"],  // p2 won
  ["p2", "p3", "p3"],  // p3 won
  ["p2", "p4", "p4"],  // p4 won
  ["p3", "p4", "p4"]   // p4 won
]

Output: ["p2", "p5"]

Explanation:
- p5 lost to p2, and p2 lost to everyone else → p5 is definitely last
- p2 won against p5 and lost to p3, p4 → p2 is definitely second to last
```

### C++ Solution
```cpp
class Solution {
public:
    vector<string> findDefiniteRankings(vector<vector<string>>& matches) {
        unordered_set<string> players;
        unordered_map<string, int> wins;
        unordered_map<string, int> losses;
        unordered_map<string, unordered_set<string>> defeatedBy;
        unordered_map<string, unordered_set<string>> defeated;

        // Build graph
        for (auto& match : matches) {
            string p1 = match[0];
            string p2 = match[1];
            string winner = match[2];

            players.insert(p1);
            players.insert(p2);

            if (winner == p1) {
                wins[p1]++;
                losses[p2]++;
                defeated[p1].insert(p2);
                defeatedBy[p2].insert(p1);
            } else {
                wins[p2]++;
                losses[p1]++;
                defeated[p2].insert(p1);
                defeatedBy[p1].insert(p2);
            }
        }

        int n = players.size();
        vector<string> result;

        for (const string& player : players) {
            // Check if rank is definite
            // Last place: lost to everyone
            if (losses[player] == n - 1) {
                result.push_back(player);
                continue;
            }

            // First place: won against everyone
            if (wins[player] == n - 1) {
                result.push_back(player);
                continue;
            }

            // Check if player's position is sandwiched
            // i.e., there's a clear set of players above and below
            unordered_set<string> above = defeatedBy[player];
            unordered_set<string> below = defeated[player];

            // If everyone is either strictly above or strictly below
            int accountedFor = above.size() + below.size() + 1; // +1 for self
            if (accountedFor == n) {
                // Further check: verify transitivity
                bool isDefinite = true;

                // Everyone above should have beaten everyone below
                for (const string& upperPlayer : above) {
                    for (const string& lowerPlayer : below) {
                        if (defeated[upperPlayer].find(lowerPlayer) == defeated[upperPlayer].end()) {
                            isDefinite = false;
                            break;
                        }
                    }
                    if (!isDefinite) break;
                }

                if (isDefinite) {
                    result.push_back(player);
                }
            }
        }

        return result;
    }
};
```

## 10. Minimum Distance to Favorite Cities

### Problem Statement
Given a directed graph of cities where edges represent distances, find the minimum distance needed to travel from a current city to any favorite city.

**Follow-up:** Find minimum distance where you must pass through a specific city en route.

### Example
```
Input:
graph = {
  "A": [("B", 5), ("C", 3)],
  "B": [("D", 2)],
  "C": [("D", 6)],
  "D": []
}
current = "A"
favorites = ["D"]

Output: 7 (A → B → D)

Follow-up:
mustPass = "C"
Output: 9 (A → C → D)
```

### C++ Solution
```cpp
class Solution {
public:
    int minDistanceToFavorites(unordered_map<string, vector<pair<string, int>>>& graph,
                                string current,
                                unordered_set<string>& favorites) {
        unordered_map<string, int> dist;
        priority_queue<pair<int, string>, vector<pair<int, string>>, greater<>> pq;

        dist[current] = 0;
        pq.push({0, current});

        while (!pq.empty()) {
            auto [d, city] = pq.top();
            pq.pop();

            if (d > dist[city]) continue;

            if (favorites.find(city) != favorites.end()) {
                return d;
            }

            for (auto& [nextCity, edgeDist] : graph[city]) {
                int newDist = d + edgeDist;

                if (dist.find(nextCity) == dist.end() || newDist < dist[nextCity]) {
                    dist[nextCity] = newDist;
                    pq.push({newDist, nextCity});
                }
            }
        }

        return -1; // No favorite city reachable
    }

    // Follow-up: must pass through specific city
    int minDistanceViaCity(unordered_map<string, vector<pair<string, int>>>& graph,
                           string current,
                           unordered_set<string>& favorites,
                           string mustPass) {
        // Distance from current to mustPass
        int dist1 = minDistanceToFavorites(graph, current, {mustPass});
        if (dist1 == -1) return -1;

        // Distance from mustPass to any favorite
        int dist2 = minDistanceToFavorites(graph, mustPass, favorites);
        if (dist2 == -1) return -1;

        return dist1 + dist2;
    }
};

// DFS version with cycle detection
class SolutionDFS {
private:
    int minDist;

public:
    int minDistanceToFavorites(unordered_map<string, vector<pair<string, int>>>& graph,
                                string current,
                                unordered_set<string>& favorites) {
        minDist = INT_MAX;
        unordered_set<string> visited;
        dfs(graph, current, favorites, visited, 0);
        return minDist == INT_MAX ? -1 : minDist;
    }

private:
    void dfs(unordered_map<string, vector<pair<string, int>>>& graph,
             string city,
             unordered_set<string>& favorites,
             unordered_set<string>& visited,
             int currentDist) {

        if (favorites.find(city) != favorites.end()) {
            minDist = min(minDist, currentDist);
            return;
        }

        visited.insert(city);

        for (auto& [nextCity, dist] : graph[city]) {
            if (visited.find(nextCity) == visited.end()) {
                dfs(graph, nextCity, favorites, visited, currentDist + dist);
            }
        }

        visited.erase(city);
    }
};
```

## 11. Max Length Subarray with Key and K Differences

### Problem Statement
Find the maximum length subarray containing only a specific key, with variations:
1. Simple: subarray with only the key
2. Multiple queries: handle multiple keys
3. With K differences: allow at most K different elements

### Example
```
Input: arr = [1, 2, 2, 1, 3, 4, 5, 2, 3, 2, 2, 2]

Query 1: key = 2
Output: 3 (last three elements [2, 2, 2])

Query 2: keys = [1, 2, 3]
Output: [1, 3, 1]

Query 3: key = 2, k = 2
Output: 6 ([2, 3, 2, 2, 2] with 2 different elements)
```

### C++ Solution
```cpp
class Solution {
public:
    // Simple: max length subarray with only key
    int maxLengthWithKey(vector<int>& arr, int key) {
        int maxLen = 0, currentLen = 0;

        for (int num : arr) {
            if (num == key) {
                currentLen++;
                maxLen = max(maxLen, currentLen);
            } else {
                currentLen = 0;
            }
        }

        return maxLen;
    }

    // Multiple queries
    vector<int> maxLengthMultipleKeys(vector<int>& arr, vector<int>& keys) {
        vector<int> result;

        for (int key : keys) {
            result.push_back(maxLengthWithKey(arr, key));
        }

        return result;
    }

    // With K different elements allowed
    int maxLengthWithKDifferences(vector<int>& arr, int key, int k) {
        int maxLen = 0;
        int left = 0;
        unordered_map<int, int> freq;
        int differentCount = 0;

        for (int right = 0; right < arr.size(); right++) {
            // Add current element
            if (arr[right] != key) {
                if (freq[arr[right]] == 0) {
                    differentCount++;
                }
                freq[arr[right]]++;
            }

            // Shrink window if too many different elements
            while (differentCount > k) {
                if (arr[left] != key) {
                    freq[arr[left]]--;
                    if (freq[arr[left]] == 0) {
                        differentCount--;
                    }
                }
                left++;
            }

            maxLen = max(maxLen, right - left + 1);
        }

        return maxLen;
    }
};
```

## 12. N-ary Tree String Decoder

### Problem Statement
Given an N-ary tree where each leaf is linked to a decoded string, decode an input string. Each path from root to leaf represents an encoding.

### Example
```
Tree:
    A
   / \
  B   C
 /   / \
Hi  D   E
    |   |
    I   am

Encodings:
"AB" → "Hi"
"ACD" → "I"
"ACE" → "am"

Input: "ABABACDACE"
Output: "HiHiIam"

Input: "ABCBACE"
Output: "Invalid"
```

### C++ Solution
```cpp
class TreeNode {
public:
    char val;
    string decoded;
    unordered_map<char, TreeNode*> children;

    TreeNode(char c) : val(c), decoded("") {}
};

class StringDecoder {
private:
    TreeNode* root;

public:
    StringDecoder(TreeNode* r) : root(r) {}

    string decode(string encoded) {
        string result = "";
        int i = 0;

        while (i < encoded.size()) {
            TreeNode* node = root;
            int start = i;

            // Try to match a path
            while (i < encoded.size()) {
                char c = encoded[i];

                if (node->children.find(c) == node->children.end()) {
                    // Invalid path
                    return "Invalid";
                }

                node = node->children[c];
                i++;

                // Check if we reached a leaf
                if (!node->decoded.empty()) {
                    result += node->decoded;
                    break;
                }

                // If we're at a node with no children and no decoded value
                if (node->children.empty() && node->decoded.empty()) {
                    return "Invalid";
                }
            }

            // If we went through all characters without finding a leaf
            if (i == encoded.size() && node->decoded.empty()) {
                return "Invalid";
            }
        }

        return result;
    }
};

// Example usage
class Solution {
public:
    string decodeString(string encoded) {
        // Build the example tree
        TreeNode* root = new TreeNode('A');

        TreeNode* b = new TreeNode('B');
        b->decoded = "Hi";
        root->children['B'] = b;

        TreeNode* c = new TreeNode('C');
        root->children['C'] = c;

        TreeNode* d = new TreeNode('D');
        d->decoded = "I";
        c->children['D'] = d;

        TreeNode* e = new TreeNode('E');
        e->decoded = "am";
        c->children['E'] = e;

        StringDecoder decoder(root);
        return decoder.decode(encoded);
    }
};
```

## 13. Sort Array by First 28 Most Significant Bits

### Problem Statement
Given an array where elements are already sorted by their first 28 most significant bits, sort the entire array efficiently.

### Example
```
Input: [1, 3, 2, 4, 10, 9]
Output: [1, 2, 3, 4, 9, 10]

Explanation:
The first 28 MSBs are all zeros for small numbers, so they're "sorted" by that criterion.
We need to fully sort considering all 32 bits.
```

### C++ Solution
```cpp
class Solution {
public:
    // Counting sort based on last 4 bits (since first 28 are sorted)
    vector<int> sortByLast4Bits(vector<int>& arr) {
        int n = arr.size();
        vector<int> result;

        // Group by first 28 bits
        unordered_map<int, vector<int>> groups;

        for (int num : arr) {
            int key = num >> 4; // First 28 bits
            groups[key].push_back(num);
        }

        // Sort each group (they differ only in last 4 bits)
        for (auto& [key, nums] : groups) {
            sort(nums.begin(), nums.end());
            for (int num : nums) {
                result.push_back(num);
            }
        }

        return result;
    }

    // Optimized: Counting sort on last 4 bits within each group
    vector<int> sortOptimized(vector<int>& arr) {
        vector<int> result;

        int i = 0;
        while (i < arr.size()) {
            int prefix = arr[i] >> 4;
            int j = i;

            // Find all elements with same 28-bit prefix
            vector<int> group;
            while (j < arr.size() && (arr[j] >> 4) == prefix) {
                group.push_back(arr[j]);
                j++;
            }

            // Sort this group (small group, simple sort is fine)
            sort(group.begin(), group.end());

            for (int num : group) {
                result.push_back(num);
            }

            i = j;
        }

        return result;
    }

    // Counting sort version (most efficient for this specific problem)
    vector<int> sortWithCountingSort(vector<int>& arr) {
        vector<int> result;

        int i = 0;
        while (i < arr.size()) {
            int prefix = arr[i] >> 4;

            // Count elements with each last 4 bits value
            vector<int> count(16, 0);
            int j = i;

            while (j < arr.size() && (arr[j] >> 4) == prefix) {
                int last4Bits = arr[j] & 15; // Get last 4 bits
                count[last4Bits]++;
                j++;
            }

            // Generate sorted elements
            for (int bit = 0; bit < 16; bit++) {
                for (int k = 0; k < count[bit]; k++) {
                    result.push_back((prefix << 4) | bit);
                }
            }

            i = j;
        }

        return result;
    }
};
```
