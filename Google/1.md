## Question 1: Maximum Value Path to Leaf Nodes

**Difficulty:** Easy

### Problem Statement
Given a binary tree, return all the leaf nodes that lie on paths with the maximum sum from root to leaf.

### Example
```
Input:
        5
       / \
      3   8
     / \   \
    2   4   9
   /
  1

Output: [9]
Explanation: Path 5->8->9 has sum 22, which is maximum
```

### C++ Solution
```cpp
class Solution {
public:
    vector<int> maxSumLeafNodes(TreeNode* root) {
        vector<int> result;
        int maxSum = INT_MIN;
        dfs(root, 0, maxSum, result);
        return result;
    }

private:
    void dfs(TreeNode* node, int currSum, int& maxSum, vector<int>& result) {
        if (!node) return;

        currSum += node->val;

        // Leaf node
        if (!node->left && !node->right) {
            if (currSum > maxSum) {
                maxSum = currSum;
                result.clear();
                result.push_back(node->val);
            } else if (currSum == maxSum) {
                result.push_back(node->val);
            }
            return;
        }

        dfs(node->left, currSum, maxSum, result);
        dfs(node->right, currSum, maxSum, result);
    }
};
```

**Time Complexity:** O(N) where N is the number of nodes
**Space Complexity:** O(H) where H is the height of the tree

## Question 2: Longest Valid Word Chain

**Difficulty:** Medium

### Problem Statement
You have a dictionary of strings. A string is "valid" if you can remove one character at a time and each intermediate string is also in the dictionary, until you reach a string of length 1. Return the length of the longest valid word.

### Example
```
Input: dict = ["string", "sring", "sing", "wording", "ing", "ng", "g"]
Output: 6
Explanation: string -> sring -> sing -> ing -> ng -> g (all present in dict)
```

### C++ Solution
```cpp
class Solution {
public:
    int longestValidWord(vector<string>& words) {
        unordered_set<string> dict(words.begin(), words.end());
        unordered_map<string, int> memo;
        int maxLen = 0;

        for (const string& word : words) {
            maxLen = max(maxLen, dfs(word, dict, memo));
        }
        return maxLen;
    }

private:
    int dfs(const string& word, unordered_set<string>& dict,
            unordered_map<string, int>& memo) {
        if (word.length() == 1) return 1;
        if (memo.count(word)) return memo[word];

        int maxChain = 1;
        for (int i = 0; i < word.length(); i++) {
            string next = word.substr(0, i) + word.substr(i + 1);
            if (dict.count(next)) {
                maxChain = max(maxChain, 1 + dfs(next, dict, memo));
            }
        }

        return memo[word] = maxChain;
    }
};
```

**Time Complexity:** O(N * L²) where N is number of words, L is max length
**Space Complexity:** O(N * L) for memoization

## Question 3: Maximum Sum of Three Connected Islands

**Difficulty:** Medium

### Problem Statement
Given N islands with values and connections between them, select a group of 3 islands A, B, C such that they are directly connected (A-B, B-C) and the sum of their values is maximum.

### Example
```
Input:
values = [1, 5, 3, 7, 2]
edges = [[0,1], [1,2], [1,3], [3,4]]

Output: 15
Explanation: Islands 1(5) -> 3(7) -> 4(2) are connected with sum = 14
             Or islands 2(3) -> 1(5) -> 3(7) with sum = 15
```

### C++ Solution
```cpp
class Solution {
public:
    int maxSumThreeIslands(vector<int>& values, vector<vector<int>>& edges) {
        int n = values.size();
        vector<vector<int>> graph(n);

        for (auto& edge : edges) {
            graph[edge[0]].push_back(edge[1]);
            graph[edge[1]].push_back(edge[0]);
        }

        int maxSum = INT_MIN;

        // For each island as middle point
        for (int mid = 0; mid < n; mid++) {
            int m = graph[mid].size();
            if (m < 2) continue;

            // Find top 2 neighbors by value
            vector<int> neighbors = graph[mid];
            sort(neighbors.begin(), neighbors.end(),
                 [&](int a, int b) { return values[a] > values[b]; });

            // Consider first two neighbors
            maxSum = max(maxSum, values[neighbors[0]] +
                         values[mid] + values[neighbors[1]]);
        }

        return maxSum;
    }
};
```

**Time Complexity:** O(N + E log E) where E is number of edges
**Space Complexity:** O(N + E)

## Question 4: Huffman Tree Level Construction

**Difficulty:** Medium

### Problem Statement
Given a map of characters to values, construct a Huffman-like binary tree where each character is placed at a node whose level corresponds to its value.

### Example
```
Input: map = {{'a', 1}, {'b', 2}, {'c', 2}, {'d', 3}}
Output: Build tree where 'a' is at level 1, 'b' and 'c' at level 2, 'd' at level 3
```

### C++ Solution
```cpp
struct TreeNode {
    char ch;
    int level;
    TreeNode *left, *right;
    TreeNode(char c, int l) : ch(c), level(l), left(nullptr), right(nullptr) {}
};

class Solution {
public:
    TreeNode* buildHuffmanTree(unordered_map<char, int>& charValues) {
        if (charValues.empty()) return nullptr;

        TreeNode* root = new TreeNode('\0', 0);

        for (auto& [ch, level] : charValues) {
            insertAtLevel(root, ch, level, 1);
        }

        return root;
    }

private:
    void insertAtLevel(TreeNode* node, char ch, int targetLevel, int currLevel) {
        if (currLevel == targetLevel) {
            // Found the level, insert here
            if (node->ch == '\0') {
                node->ch = ch;
            }
            return;
        }

        // Go deeper
        if (!node->left) node->left = new TreeNode('\0', currLevel + 1);
        if (!node->right) node->right = new TreeNode('\0', currLevel + 1);

        if (node->left->ch == '\0') {
            insertAtLevel(node->left, ch, targetLevel, currLevel + 1);
        } else {
            insertAtLevel(node->right, ch, targetLevel, currLevel + 1);
        }
    }
};
```

**Time Complexity:** O(N * H) where N is chars, H is max level
**Space Complexity:** O(N)

## Question 5: Stock Value Tracker with Updates

**Difficulty:** Medium

### Problem Statement
Design a data structure to store stock values (MIN/MAX/Current) that change every millisecond. Support querying these values at any timestamp and updating previous values.

### Example
```
Operations:
addValue(100, timestamp=1000)
addValue(150, timestamp=1001)
getMin(1000, 1001) -> 100
update(1000, 120)
getMin(1000, 1001) -> 120
```

### C++ Solution
```cpp
class StockTracker {
private:
    map<long long, int> values; // timestamp -> value

public:
    void addValue(int value, long long timestamp) {
        values[timestamp] = value;
    }

    void updateValue(long long timestamp, int newValue) {
        if (values.count(timestamp)) {
            values[timestamp] = newValue;
        }
    }

    int getMin(long long start, long long end) {
        auto it_start = values.lower_bound(start);
        auto it_end = values.upper_bound(end);

        int minVal = INT_MAX;
        for (auto it = it_start; it != it_end; ++it) {
            minVal = min(minVal, it->second);
        }
        return minVal;
    }

    int getMax(long long start, long long end) {
        auto it_start = values.lower_bound(start);
        auto it_end = values.upper_bound(end);

        int maxVal = INT_MIN;
        for (auto it = it_start; it != it_end; ++it) {
            maxVal = max(maxVal, it->second);
        }
        return maxVal;
    }

    int getCurrent(long long timestamp) {
        if (values.count(timestamp)) {
            return values[timestamp];
        }
        return -1; // or throw exception
    }
};
```

**Time Complexity:**
- addValue/update: O(log N)
- getMin/getMax: O(K) where K is values in range

**Space Complexity:** O(N)

## Question 6: Photo Upload Tracker

**Difficulty:** Medium

### Problem Statement
Photos are uploaded to a server in any order (1 to N). When acknowledgement for photo K is received, find the maximum photo number that has been uploaded continuously from photo 1.

### Example
```
ack(1) -> getMax() returns 1
ack(2) -> getMax() returns 2
ack(4) -> getMax() returns 2 (3 missing)
ack(5) -> getMax() returns 2 (3 missing)
ack(3) -> getMax() returns 5 (1,2,3,4,5 all received)
```

### C++ Solution
```cpp
class PhotoUploadTracker {
private:
    unordered_set<int> uploaded;
    int maxContinuous;

public:
    PhotoUploadTracker() : maxContinuous(0) {}

    void ack(int photoNum) {
        uploaded.insert(photoNum);

        // Update maxContinuous if this extends the sequence
        while (uploaded.count(maxContinuous + 1)) {
            maxContinuous++;
        }
    }

    int getMax() {
        return maxContinuous;
    }
};
```

**Time Complexity:** O(1) amortized for both operations
**Space Complexity:** O(N)

## Question 7: Paint Line Segments

**Difficulty:** Medium

### Problem Statement
Given a line (X-axis) and pairs of coordinates to paint like (4,10), (7,13), calculate the cost to paint each segment. Cost equals distance. Already painted areas don't need repainting.

### Example
```
Input: segments = [[4,10], [7,13], [16,20], [1,40]]
Output: [6, 3, 4, 26]
Explanation:
- [4,10]: cost = 10-4 = 6
- [7,13]: [7,10] already painted, cost = 13-10 = 3
- [16,20]: cost = 20-16 = 4
- [1,40]: cost = 39 - (already painted lengths) = 26
```

### C++ Solution
```cpp
class Solution {
public:
    vector<int> paintCosts(vector<vector<int>>& segments) {
        vector<int> result;
        vector<pair<int,int>> painted; // merged intervals

        for (auto& seg : segments) {
            int start = seg[0], end = seg[1];
            int cost = end - start;

            vector<pair<int,int>> newPainted;
            bool merged = false;

            for (auto& p : painted) {
                if (end < p.first || start > p.second) {
                    newPainted.push_back(p);
                } else {
                    // Overlap found
                    int overlapStart = max(start, p.first);
                    int overlapEnd = min(end, p.second);
                    cost -= (overlapEnd - overlapStart);

                    start = min(start, p.first);
                    end = max(end, p.second);
                    merged = true;
                }
            }

            newPainted.push_back({start, end});
            sort(newPainted.begin(), newPainted.end());

            // Merge overlapping intervals
            painted.clear();
            for (auto& p : newPainted) {
                if (painted.empty() || painted.back().second < p.first) {
                    painted.push_back(p);
                } else {
                    painted.back().second = max(painted.back().second, p.second);
                }
            }

            result.push_back(cost);
        }

        return result;
    }
};
```

**Time Complexity:** O(N²) where N is number of segments
**Space Complexity:** O(N)

## Question 8: Count Houses in Circular Street

**Difficulty:** Medium

### Problem Statement
Houses are arranged in a circle. You start at a random house with all doors closed. Using operations `openDoor()`, `closeDoor()`, `isDoorOpen()`, `moveRight()`, `moveLeft()`, count total houses.

**Follow-up:** Doors are randomly open/closed. Count houses with constraint: total houses ≤ max.

### C++ Solution
```cpp
class Solution {
public:
    // Part 1: All doors initially closed
    int countHouses() {
        int count = 1;
        openDoor();
        moveRight();

        while (!isDoorOpen()) {
            count++;
            moveRight();
        }

        return count;
    }

    // Part 2: Random initial state
    int countHousesRandom(int maxHouses) {
        // Mark starting position uniquely
        bool wasOpen = isDoorOpen();
        openDoor();
        moveRight();
        closeDoor();
        moveRight();

        int count = 2;

        // Travel until we find: open followed by closed (our marker)
        while (count <= maxHouses) {
            if (isDoorOpen()) {
                moveRight();
                if (!isDoorOpen()) {
                    // Found our marker
                    break;
                }
                moveRight();
            } else {
                moveRight();
            }
            count++;
        }

        // Restore initial state
        moveLeft();
        if (wasOpen) openDoor();

        return count;
    }
};
```

**Time Complexity:** O(N) where N is number of houses
**Space Complexity:** O(1)

## Question 9: Hospital Room Allocation

**Difficulty:** Medium

### Problem Statement
A hospital has N rooms (1 to N). Patients arrive with start time and treatment duration. Rooms are allocated by lowest available index. Find which room served the maximum number of patients.

### Example
```
Input: N = 2, patients = [{start:1, duration:8}, {start:1, duration:2}, {start:6, duration:4}]
Output: 2
Explanation:
- Patient 1 -> Room 1 (time 1-9)
- Patient 2 -> Room 2 (time 1-3)
- Patient 3 -> Room 2 (time 6-10) [Room 2 free at time 3]
Room 2 served 2 patients (max)
```

### C++ Solution
```cpp
class Solution {
public:
    int maxPatientsRoom(int N, vector<pair<int,int>>& patients) {
        // {end_time, room_number}
        priority_queue<pair<int,int>, vector<pair<int,int>>, greater<>> pq;
        vector<int> roomCount(N + 1, 0);

        // Initialize all rooms
        for (int i = 1; i <= N; i++) {
            pq.push({0, i});
        }

        for (auto& patient : patients) {
            int start = patient.first;
            int duration = patient.second;

            auto [endTime, room] = pq.top();
            pq.pop();

            roomCount[room]++;
            int newEndTime = max(endTime, start) + duration;
            pq.push({newEndTime, room});
        }

        int maxRoom = 1;
        for (int i = 1; i <= N; i++) {
            if (roomCount[i] > roomCount[maxRoom]) {
                maxRoom = i;
            }
        }

        return maxRoom;
    }
};
```

**Time Complexity:** O(P log N) where P is patients, N is rooms
**Space Complexity:** O(N)

## Question 10: Tree Category Coupon Compatibility

**Difficulty:** Hard

### Problem Statement
Given a tree structure storing product categories and regions, check coupon compatibility by reaching leaf nodes and verifying against default coupons. Each product is a category and each region is a continent.

### Example
```
Input: Tree with categories and regions, default coupons list
Output: Boolean indicating if coupon is compatible
```

### C++ Solution
```cpp
struct CategoryNode {
    string name;
    vector<CategoryNode*> children;
    bool isLeaf;

    CategoryNode(string n) : name(n), isLeaf(false) {}
};

class Solution {
public:
    bool isCouponCompatible(CategoryNode* root,
                           unordered_set<string>& defaultCoupons,
                           string targetCategory) {
        vector<string> leafCategories;
        collectLeaves(root, leafCategories);

        for (const string& leaf : leafCategories) {
            if (leaf == targetCategory && defaultCoupons.count(leaf)) {
                return true;
            }
        }

        return false;
    }

private:
    void collectLeaves(CategoryNode* node, vector<string>& leaves) {
        if (!node) return;

        if (node->isLeaf || node->children.empty()) {
            leaves.push_back(node->name);
            return;
        }

        for (CategoryNode* child : node->children) {
            collectLeaves(child, leaves);
        }
    }
};
```

**Time Complexity:** O(N) where N is nodes in tree
**Space Complexity:** O(H) where H is height

## Tips for Google Interviews

1. **Start with clarifying questions** - Understand constraints and edge cases
2. **Think out loud** - Communicate your thought process
3. **Start with brute force** - Then optimize
4. **Test your code** - Walk through with examples
5. **Analyze complexity** - Be ready to discuss time/space trade-offs
6. **Practice common patterns** - Trees, Graphs, DP, Binary Search, Sliding Window
7. **Know your data structures** - When to use HashMap, TreeMap, PriorityQueue, etc.
