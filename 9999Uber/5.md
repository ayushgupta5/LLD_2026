## 📝 DSA Problems & Solutions

### 1. Merge K Cars Problem

**Problem:** Given car names, merge K identical consecutive cars. Repeat until no more merges possible.

**Example:** `[Honda, Honda, Maruti, Maruti, Maruti, BMW, Maruti, Maruti, Maruti]`, K=2
**Output:** `[Honda, Maruti, BMW, Maruti]`

```cpp
vector<string> mergeKCars(vector<string>& cars, int K) {
    bool merged = true;
    
    while (merged) {
        merged = false;
        vector<string> result;
        int i = 0;
        
        while (i < cars.size()) {
            int count = 1;
            string current = cars[i];
            
            // Count consecutive identical cars
            while (i + count < cars.size() && cars[i + count] == current) {
                count++;
            }
            
            // If we can merge K cars
            if (count >= K) {
                int remaining = count % K;
                if (remaining > 0) {
                    result.push_back(current);
                }
                merged = true;
            } else {
                // Add all cars that can't be merged
                for (int j = 0; j < count; j++) {
                    result.push_back(current);
                }
            }
            i += count;
        }
        cars = result;
    }
    return cars;
}
```

**Time:** O(N * iterations), **Space:** O(N)

**Key Points:**
- Keep merging until no changes occur
- Count consecutive identical elements
- Remove K elements at a time, keep remainder

### 2. Line Sweep Algorithm (Generic Template)

**Common Pattern:** Events with start/end times, overlapping intervals

```cpp
// Template for line sweep problems
int lineSweep(vector<pair<int,int>>& intervals) {
    vector<pair<int, int>> events; // {position, type: +1 start, -1 end}
    
    for (auto& interval : intervals) {
        events.push_back({interval.first, 1});
        events.push_back({interval.second, -1});
    }
    
    sort(events.begin(), events.end());
    
    int maxOverlap = 0, currentCount = 0;
    for (auto& event : events) {
        currentCount += event.second;
        maxOverlap = max(maxOverlap, currentCount);
    }
    
    return maxOverlap;
}
```

**Use Cases:** Meeting rooms, interval overlaps, skyline problem

### 3. Base 2 to Base 6 Conversion

**Problem:** Convert binary string (up to 100 bits) to base 6

```cpp
string sumBaseB(string a, string b, int base) {
    string result = "";
    int carry = 0;
    int i = a.length() - 1, j = b.length() - 1;
    
    while (i >= 0 || j >= 0 || carry) {
        int sum = carry;
        if (i >= 0) sum += (a[i--] - '0');
        if (j >= 0) sum += (b[j--] - '0');
        
        result = char('0' + sum % base) + result;
        carry = sum / base;
    }
    
    return result;
}

string convertBase2To6(string binary) {
    string ans = "0";
    
    for (char bit : binary) {
        // ans = ans * 2 (in base 6)
        ans = sumBaseB(ans, ans, 6);
        
        // Add current bit (0 or 1) in base 6
        if (bit == '1') {
            ans = sumBaseB(ans, "1", 6);
        }
    }
    
    return ans;
}
```

**Time:** O(N²), **Space:** O(N)

**Key Insight:** Process bit by bit, doubling and adding in base 6

### 4. Binary String - Maximum K (Deque Approach)

**Problem:** Make all bits equal by removing substrings. Find max K such that substring length > K.

```cpp
int maxKToMakeUniform(string s) {
    deque<int> dq;
    int n = s.length();
    
    // Store lengths of consecutive identical characters
    int count = 1;
    for (int i = 1; i < n; i++) {
        if (s[i] == s[i-1]) {
            count++;
        } else {
            dq.push_back(count);
            count = 1;
        }
    }
    dq.push_back(count);
    
    int answer = n;
    
    // Merge segments from ends until one remains
    while (dq.size() > 1) {
        int minVal = min(dq.front(), dq.back());
        answer = min(answer, n - minVal);
        
        if (dq.front() < dq.back()) {
            int val = dq.front();
            dq.pop_front();
            dq.front() += val;
        } else {
            int val = dq.back();
            dq.pop_back();
            dq.back() += val;
        }
    }
    
    answer = min(answer, n - dq.front());
    return answer;
}
```

**Alternative - Binary Search:**
```cpp
bool canMakeUniform(string s, int k) {

int maxKBinarySearch(string s) {
```

### 5. Bowling Pins (Burst Balloons Variant)

**Problem:** Knock down pairs of consecutive pins for max points. Score = arr[i-1] * arr[i] * arr[i+1] * arr[i+2]

```cpp
const int MOD = 1e9 + 7;

int maxPoints(vector<int>& arr) {
    int n = arr.size();
    
    // Add boundary pins with value 1
    vector<int> pins(n + 2, 1);
    for (int i = 0; i < n; i++) {
        pins[i + 1] = arr[i];
    }
    
    // dp[i][j] = max points from knocking pins in range (i, j)
    vector<vector<long long>> dp(n + 2, vector<long long>(n + 2, 0));
    
    // len = number of pins between i and j
    for (int len = 1; len <= n; len++) {
        for (int i = 0; i + len + 1 < pins.size(); i++) {
            int j = i + len + 1;
            
            // Try knocking each consecutive pair (k, k+1) last
            for (int k = i + 1; k < j - 1; k++) {
                long long points = (1LL * pins[i] * pins[k] % MOD * 
                                    pins[k+1] % MOD * pins[j] % MOD);
                
                dp[i][j] = max(dp[i][j], 
                              (dp[i][k] + dp[k+1][j] + points) % MOD);
            }
        }
    }
    
    return dp[0][n + 1];
}
```

**Time:** O(N³), **Space:** O(N²)

**Key Insight:** 
- Similar to burst balloons - think "which pair to knock LAST"
- DP interval approach with boundary handling

## 🔍 Common Patterns Quick Reference

### Binary Search Template
```cpp
int binarySearch(vector<int>& arr, int target) {
    int left = 0, right = arr.size() - 1;
    
    while (left <= right) {
        int mid = left + (right - left) / 2;
        
        if (arr[mid] == target) return mid;
        else if (arr[mid] < target) left = mid + 1;
        else right = mid - 1;
    }
    return -1;
}
```

### Merge Intervals
```cpp
vector<vector<int>> merge(vector<vector<int>>& intervals) {
    sort(intervals.begin(), intervals.end());
    vector<vector<int>> merged;
    
    for (auto& interval : intervals) {
        if (merged.empty() || merged.back()[1] < interval[0]) {
            merged.push_back(interval);
        } else {
            merged.back()[1] = max(merged.back()[1], interval[1]);
        }
    }
    return merged;
}
```

### Graph BFS Template
```cpp
void bfs(vector<vector<int>>& graph, int start) {
    queue<int> q;
    vector<bool> visited(graph.size(), false);
    
    q.push(start);
    visited[start] = true;
    
    while (!q.empty()) {
        int node = q.front();
        q.pop();
        
        for (int neighbor : graph[node]) {
            if (!visited[neighbor]) {
                visited[neighbor] = true;
                q.push(neighbor);
            }
        }
    }
}
```

### Graph DFS Template
```cpp
void dfs(vector<vector<int>>& graph, int node, vector<bool>& visited) {
    visited[node] = true;
    
    for (int neighbor : graph[node]) {
        if (!visited[neighbor]) {
            dfs(graph, neighbor, visited);
        }
    }
}
```

### Sliding Window Template
```cpp
int slidingWindow(vector<int>& arr, int k) {
    int left = 0, right = 0;
    int maxSum = 0, currentSum = 0;
    
    while (right < arr.size()) {
        currentSum += arr[right];
        
        while (right - left + 1 > k) {
            currentSum -= arr[left];
            left++;
        }
        
        if (right - left + 1 == k) {
            maxSum = max(maxSum, currentSum);
        }
        right++;
    }
    return maxSum;
}
```

### Two Pointers Template
```cpp
vector<int> twoSum(vector<int>& arr, int target) {
    int left = 0, right = arr.size() - 1;
    
    while (left < right) {
        int sum = arr[left] + arr[right];
        
        if (sum == target) {
            return {left, right};
        } else if (sum < target) {
            left++;
        } else {
            right--;
        }
    }
    return {-1, -1};
}
```

## 🎯 Graph Algorithms

### Dijkstra's Algorithm
```cpp
vector<int> dijkstra(vector<vector<pair<int,int>>>& graph, int start) {
    int n = graph.size();
    vector<int> dist(n, INT_MAX);
    priority_queue<pair<int,int>, vector<pair<int,int>>, greater<>> pq;
    
    dist[start] = 0;
    pq.push({0, start});
    
    while (!pq.empty()) {
        auto [d, u] = pq.top();
        pq.pop();
        
        if (d > dist[u]) continue;
        
        for (auto [v, weight] : graph[u]) {
            if (dist[u] + weight < dist[v]) {
                dist[v] = dist[u] + weight;
                pq.push({dist[v], v});
            }
        }
    }
    return dist;
}
```

### Topological Sort (Kahn's Algorithm)
```cpp
vector<int> topologicalSort(int n, vector<vector<int>>& graph) {
    vector<int> inDegree(n, 0);
    
    for (int u = 0; u < n; u++) {
        for (int v : graph[u]) {
            inDegree[v]++;
        }
    }
    
    queue<int> q;
    for (int i = 0; i < n; i++) {
        if (inDegree[i] == 0) q.push(i);
    }
    
    vector<int> result;
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        result.push_back(u);
        
        for (int v : graph[u]) {
            if (--inDegree[v] == 0) {
                q.push(v);
            }
        }
    }
    
    return result.size() == n ? result : vector<int>();
}
```

### Union-Find (Disjoint Set)
```cpp
class UnionFind {
    vector<int> parent, rank;
public:
    UnionFind(int n) : parent(n), rank(n, 0) {
        iota(parent.begin(), parent.end(), 0);
    }
    
    int find(int x) {
        if (parent[x] != x) {
            parent[x] = find(parent[x]); // Path compression
        }
        return parent[x];
    }
    
    bool unite(int x, int y) {
        int px = find(x), py = find(y);
        if (px == py) return false;
        
        // Union by rank
        if (rank[px] < rank[py]) swap(px, py);
        parent[py] = px;
        if (rank[px] == rank[py]) rank[px]++;
        
        return true;
    }
};
```
