### 1. Find the Closest Palindrome (19 times)
**Problem:** Given a number N, find the smallest palindrome greater than N.

**Approach:** Handle edge cases systematically
- Single digit: return n+1
- All 9s: return 100...001
- Mirror first half to second half
- If mirrored ≤ n, increment middle and mirror again

```cpp
class Solution {
public:
    string nearestPalindromic(string n) {
        int len = n.length();
        set<long long> candidates;

        // Edge cases
        candidates.insert((long long)pow(10, len) + 1);  // 100...001
        candidates.insert((long long)pow(10, len - 1) - 1);  // 99...99

        // Get prefix and create palindrome
        long long prefix = stoll(n.substr(0, (len + 1) / 2));

        for (int i = -1; i <= 1; i++) {
            string palin = toPalindrome(to_string(prefix + i), len % 2 == 0);
            candidates.insert(stoll(palin));
        }

        candidates.erase(stoll(n));
        long long num = stoll(n);
        long long minDiff = LLONG_MAX, result = 0;

        for (long long cand : candidates) {
            long long diff = abs(cand - num);
            if (diff < minDiff || (diff == minDiff && cand < result)) {
                minDiff = diff;
                result = cand;
            }
        }
        return to_string(result);
    }

private:
    string toPalindrome(string half, bool even) {
        string res = half;
        if (!even) half.pop_back();
        reverse(half.begin(), half.end());
        return res + half;
    }
};
```
**Time:** O(1), **Space:** O(1)

### 2. Bus Routes (40 times)
**Problem:** Find minimum buses needed to reach target stop from source.

**Approach:** BFS on bus routes (not stops)
- Treat each bus route as a node
- Build graph of connected routes
- BFS from routes containing source to routes containing target

```cpp
class Solution {
public:
    int numBusesToDestination(vector<vector<int>>& routes, int source, int target) {
        if (source == target) return 0;

        unordered_map<int, vector<int>> stopToRoutes;
        for (int i = 0; i < routes.size(); i++) {
            for (int stop : routes[i]) {
                stopToRoutes[stop].push_back(i);
            }
        }

        queue<int> q;
        unordered_set<int> visitedRoutes;

        for (int route : stopToRoutes[source]) {
            q.push(route);
            visitedRoutes.insert(route);
        }

        int buses = 1;
        while (!q.empty()) {
            int size = q.size();
            for (int i = 0; i < size; i++) {
                int route = q.front(); q.pop();

                for (int stop : routes[route]) {
                    if (stop == target) return buses;

                    for (int nextRoute : stopToRoutes[stop]) {
                        if (visitedRoutes.find(nextRoute) == visitedRoutes.end()) {
                            visitedRoutes.insert(nextRoute);
                            q.push(nextRoute);
                        }
                    }
                }
            }
            buses++;
        }
        return -1;
    }
};
```
**Time:** O(N*K) where N=routes, K=stops per route, **Space:** O(N*K)

### 3. Longest Continuous Subarray With Absolute Diff ≤ Limit (15 times)
**Problem:** Find longest subarray where |max - min| ≤ limit.

**Approach:** Sliding window with two deques (max and min tracking)

```cpp
class Solution {
public:
    int longestSubarray(vector<int>& nums, int limit) {
        deque<int> maxDeque, minDeque;
        int left = 0, result = 0;

        for (int right = 0; right < nums.size(); right++) {
            // Maintain decreasing deque for max
            while (!maxDeque.empty() && nums[right] > maxDeque.back()) {
                maxDeque.pop_back();
            }
            maxDeque.push_back(nums[right]);

            // Maintain increasing deque for min
            while (!minDeque.empty() && nums[right] < minDeque.back()) {
                minDeque.pop_back();
            }
            minDeque.push_back(nums[right]);

            // Shrink window if diff exceeds limit
            while (maxDeque.front() - minDeque.front() > limit) {
                if (nums[left] == maxDeque.front()) maxDeque.pop_front();
                if (nums[left] == minDeque.front()) minDeque.pop_front();
                left++;
            }

            result = max(result, right - left + 1);
        }
        return result;
    }
};
```
**Time:** O(N), **Space:** O(N)

### 4. Leftmost Column with at Least a One (15 times)
**Problem:** Find leftmost column with a 1 in row-sorted binary matrix.

**Approach:** Start from top-right, move left on 1, move down on 0

```cpp
class Solution {
public:
    int leftMostColumnWithOne(BinaryMatrix &binaryMatrix) {
        auto dim = binaryMatrix.dimensions();
        int rows = dim[0], cols = dim[1];

        int row = 0, col = cols - 1;
        int result = -1;

        while (row < rows && col >= 0) {
            if (binaryMatrix.get(row, col) == 1) {
                result = col;
                col--;  // Move left
            } else {
                row++;  // Move down
            }
        }
        return result;
    }
};
```
**Time:** O(M+N), **Space:** O(1)

### 5. Merge Intervals (13 times)
**Problem:** Merge overlapping intervals.

```cpp
class Solution {
public:
    vector<vector<int>> merge(vector<vector<int>>& intervals) {
        if (intervals.empty()) return {};

        sort(intervals.begin(), intervals.end());
        vector<vector<int>> result = {intervals[0]};

        for (int i = 1; i < intervals.size(); i++) {
            if (intervals[i][0] <= result.back()[1]) {
                result.back()[1] = max(result.back()[1], intervals[i][1]);
            } else {
                result.push_back(intervals[i]);
            }
        }
        return result;
    }
};
```
**Time:** O(N log N), **Space:** O(1)

### 6. Number of Islands (12 times)
**Problem:** Count islands in a 2D grid.

**Approach:** DFS/BFS to mark connected components

```cpp
class Solution {
public:
    int numIslands(vector<vector<char>>& grid) {
        if (grid.empty()) return 0;
        int count = 0;

        for (int i = 0; i < grid.size(); i++) {
            for (int j = 0; j < grid[0].size(); j++) {
                if (grid[i][j] == '1') {
                    count++;
                    dfs(grid, i, j);
                }
            }
        }
        return count;
    }

private:
    void dfs(vector<vector<char>>& grid, int i, int j) {
        if (i < 0 || i >= grid.size() || j < 0 || j >= grid[0].size() ||
            grid[i][j] != '1') return;

        grid[i][j] = '0';
        dfs(grid, i+1, j);
        dfs(grid, i-1, j);
        dfs(grid, i, j+1);
        dfs(grid, i, j-1);
    }
};
```
**Time:** O(M*N), **Space:** O(M*N) for recursion

### 7. Word Search II (12 times)
**Problem:** Find all words from dictionary in 2D board.

**Approach:** Trie + DFS backtracking

```cpp
struct TrieNode {
    unordered_map<char, TrieNode*> children;
    string word;
};

class Solution {
public:
    vector<string> findWords(vector<vector<char>>& board, vector<string>& words) {
        TrieNode* root = buildTrie(words);
        vector<string> result;

        for (int i = 0; i < board.size(); i++) {
            for (int j = 0; j < board[0].size(); j++) {
                dfs(board, i, j, root, result);
            }
        }
        return result;
    }

private:
    TrieNode* buildTrie(vector<string>& words) {
        TrieNode* root = new TrieNode();
        for (string& word : words) {
            TrieNode* node = root;
            for (char c : word) {
                if (!node->children[c]) {
                    node->children[c] = new TrieNode();
                }
                node = node->children[c];
            }
            node->word = word;
        }
        return root;
    }

    void dfs(vector<vector<char>>& board, int i, int j, TrieNode* node,
             vector<string>& result) {
        if (i < 0 || i >= board.size() || j < 0 || j >= board[0].size()) return;

        char c = board[i][j];
        if (c == '#' || !node->children[c]) return;

        node = node->children[c];
        if (!node->word.empty()) {
            result.push_back(node->word);
            node->word = "";  // Avoid duplicates
        }

        board[i][j] = '#';
        dfs(board, i+1, j, node, result);
        dfs(board, i-1, j, node, result);
        dfs(board, i, j+1, node, result);
        dfs(board, i, j-1, node, result);
        board[i][j] = c;
    }
};
```
**Time:** O(M*N*4^L) where L=max word length, **Space:** O(K) K=total chars in dictionary

### 8. Evaluate Division (12 times)
**Problem:** Given equations and values, evaluate queries.

**Approach:** Build weighted graph, DFS/BFS for each query

```cpp
class Solution {
public:
    vector<double> calcEquation(vector<vector<string>>& equations,
                                vector<double>& values,
                                vector<vector<string>>& queries) {
        unordered_map<string, vector<pair<string, double>>> graph;

        // Build graph
        for (int i = 0; i < equations.size(); i++) {
            string a = equations[i][0], b = equations[i][1];
            graph[a].push_back({b, values[i]});
            graph[b].push_back({a, 1.0 / values[i]});
        }

        vector<double> result;
        for (auto& query : queries) {
            unordered_set<string> visited;
            double val = dfs(graph, query[0], query[1], visited);
            result.push_back(val);
        }
        return result;
    }

private:
    double dfs(unordered_map<string, vector<pair<string, double>>>& graph,
               string src, string dst, unordered_set<string>& visited) {
        if (graph.find(src) == graph.end()) return -1.0;
        if (src == dst) return 1.0;

        visited.insert(src);
        for (auto& [next, val] : graph[src]) {
            if (visited.find(next) != visited.end()) continue;
            double result = dfs(graph, next, dst, visited);
            if (result != -1.0) return result * val;
        }
        return -1.0;
    }
};
```
**Time:** O(Q * (V+E)) for Q queries, **Space:** O(V+E)

### 9. Kth Smallest Element in a BST (12 times)
**Problem:** Find kth smallest in BST.

**Approach:** Inorder traversal (iterative or recursive)

```cpp
class Solution {
public:
    int kthSmallest(TreeNode* root, int k) {
        stack<TreeNode*> st;
        TreeNode* curr = root;

        while (true) {
            while (curr) {
                st.push(curr);
                curr = curr->left;
            }
            curr = st.top(); st.pop();
            if (--k == 0) return curr->val;
            curr = curr->right;
        }
    }
};
```
**Time:** O(H + K), **Space:** O(H)

### 10. Word Search (11 times)
**Problem:** Find if word exists in 2D board.

```cpp
class Solution {
public:
    bool exist(vector<vector<char>>& board, string word) {
        for (int i = 0; i < board.size(); i++) {
            for (int j = 0; j < board[0].size(); j++) {
                if (dfs(board, word, i, j, 0)) return true;
            }
        }
        return false;
    }

private:
    bool dfs(vector<vector<char>>& board, string& word, int i, int j, int idx) {
        if (idx == word.size()) return true;
        if (i < 0 || i >= board.size() || j < 0 || j >= board[0].size() ||
            board[i][j] != word[idx]) return false;

        char temp = board[i][j];
        board[i][j] = '#';

        bool found = dfs(board, word, i+1, j, idx+1) ||
                     dfs(board, word, i-1, j, idx+1) ||
                     dfs(board, word, i, j+1, idx+1) ||
                     dfs(board, word, i, j-1, idx+1);

        board[i][j] = temp;
        return found;
    }
};
```
**Time:** O(M*N*4^L), **Space:** O(L)

### 11. Longest Repeating Character Replacement (10 times)
**Problem:** Max length substring after replacing k characters.

**Approach:** Sliding window with frequency tracking

```cpp
class Solution {
public:
    int characterReplacement(string s, int k) {
        vector<int> count(26, 0);
        int left = 0, maxCount = 0, result = 0;

        for (int right = 0; right < s.length(); right++) {
            maxCount = max(maxCount, ++count[s[right] - 'A']);

            // Window size - most frequent char > k replacements
            while (right - left + 1 - maxCount > k) {
                count[s[left] - 'A']--;
                left++;
            }

            result = max(result, right - left + 1);
        }
        return result;
    }
};
```
**Time:** O(N), **Space:** O(1)

### 12. Top K Frequent Words (10 times)
**Problem:** Return k most frequent words.

**Approach:** HashMap + Min Heap or Bucket Sort

```cpp
class Solution {
public:
    vector<string> topKFrequent(vector<string>& words, int k) {
        unordered_map<string, int> freq;
        for (string& word : words) freq[word]++;

        auto cmp = [](pair<string, int>& a, pair<string, int>& b) {
            return a.second > b.second ||
                   (a.second == b.second && a.first < b.first);
        };
        priority_queue<pair<string, int>, vector<pair<string, int>>,
                       decltype(cmp)> pq(cmp);

        for (auto& [word, count] : freq) {
            pq.push({word, count});
            if (pq.size() > k) pq.pop();
        }

        vector<string> result;
        while (!pq.empty()) {
            result.push_back(pq.top().first);
            pq.pop();
        }
        reverse(result.begin(), result.end());
        return result;
    }
};
```
**Time:** O(N log k), **Space:** O(N)

### 14. Insert Delete GetRandom O(1) (8 times)
**Problem:** Design data structure with O(1) operations.

```cpp
class RandomizedSet {
private:
    vector<int> nums;
    unordered_map<int, int> valToIndex;

public:
    bool insert(int val) {
        if (valToIndex.find(val) != valToIndex.end()) return false;
        nums.push_back(val);
        valToIndex[val] = nums.size() - 1;
        return true;
    }

    bool remove(int val) {
        if (valToIndex.find(val) == valToIndex.end()) return false;
        int idx = valToIndex[val];
        int lastVal = nums.back();
        nums[idx] = lastVal;
        valToIndex[lastVal] = idx;
        nums.pop_back();
        valToIndex.erase(val);
        return true;
    }

    int getRandom() {
        return nums[rand() % nums.size()];
    }
};
```
**Time:** O(1) all operations, **Space:** O(N)

### 15. Text Justification (8 times)
**Problem:** Justify text to fit line width.

```cpp
class Solution {
public:
    vector<string> fullJustify(vector<string>& words, int maxWidth) {
        vector<string> result;
        int i = 0;

        while (i < words.size()) {
            int lineLen = words[i].length();
            int j = i + 1;

            // Find words that fit in current line
            while (j < words.size() && lineLen + 1 + words[j].length() <= maxWidth) {
                lineLen += 1 + words[j].length();
                j++;
            }

            int numWords = j - i;
            int numSpaces = maxWidth - lineLen + (numWords - 1);

            string line;
            // Last line or single word
            if (j == words.size() || numWords == 1) {
                for (int k = i; k < j; k++) {
                    line += words[k];
                    if (k < j - 1) line += " ";
                }
                line += string(maxWidth - line.length(), ' ');
            } else {
                int spacesPerGap = numSpaces / (numWords - 1);
                int extraSpaces = numSpaces % (numWords - 1);

                for (int k = i; k < j; k++) {
                    line += words[k];
                    if (k < j - 1) {
                        line += string(spacesPerGap + (k - i < extraSpaces ? 1 : 0), ' ');
                    }
                }
            }

            result.push_back(line);
            i = j;
        }
        return result;
    }
};
```
**Time:** O(N), **Space:** O(1)

### 16. Minimum Time to Complete Trips (7 times)
**Problem:** Find minimum time for n trips with given cab times.

**Approach:** Binary search on time

```cpp
class Solution {
public:
    long long minimumTime(vector<int>& time, int totalTrips) {
        long long left = 1, right = 1LL * *min_element(time.begin(), time.end()) * totalTrips;

        while (left < right) {
            long long mid = left + (right - left) / 2;
            if (canComplete(time, totalTrips, mid)) {
                right = mid;
            } else {
                left = mid + 1;
            }
        }
        return left;
    }

private:
    bool canComplete(vector<int>& time, int totalTrips, long long maxTime) {
        long long trips = 0;
        for (int t : time) {
            trips += maxTime / t;
            if (trips >= totalTrips) return true;
        }
        return trips >= totalTrips;
    }
};
```
**Time:** O(N log(min*totalTrips)), **Space:** O(1)

### 17. Longest Substring Without Repeating Characters (7 times)

```cpp
class Solution {
```

### 19. Weighted Random Selection
**Problem:** Pick random number with weights.

**Approach:** Prefix sum + binary search

```cpp
class WeightedRandom {
private:
    vector<int> prefixSum;

public:
    WeightedRandom(vector<int>& weights) {
        prefixSum.resize(weights.size());
        prefixSum[0] = weights[0];
        for (int i = 1; i < weights.size(); i++) {
            prefixSum[i] = prefixSum[i-1] + weights[i];
        }
    }

    int pickIndex() {
        int target = rand() % prefixSum.back();
        return lower_bound(prefixSum.begin(), prefixSum.end(), target + 1)
               - prefixSum.begin();
    }
};
```
**Time:** O(log N) per pick, **Space:** O(N)

### 20. Minimize Transactions
**Problem:** Simplify debt settlements.

**Approach:** Calculate net balance, match positive with negative

```cpp
void minimizeTransactions(vector<tuple<string, string, int>>& transactions) {
    unordered_map<string, int> balance;

    // Calculate net balance for each person
    for (auto& [from, to, amount] : transactions) {
        balance[from] -= amount;
        balance[to] += amount;
    }

    vector<pair<string, int>> creditors, debtors;
    for (auto& [person, bal] : balance) {
        if (bal > 0) creditors.push_back({person, bal});
        else if (bal < 0) debtors.push_back({person, -bal});
    }

    // Match creditors with debtors
    vector<tuple<string, string, int>> result;
    int i = 0, j = 0;

    while (i < creditors.size() && j < debtors.size()) {
        int amount = min(creditors[i].second, debtors[j].second);
        result.push_back({debtors[j].first, creditors[i].first, amount});

        creditors[i].second -= amount;
        debtors[j].second -= amount;

        if (creditors[i].second == 0) i++;
        if (debtors[j].second == 0) j++;
    }
}
```
**Time:** O(N), **Space:** O(N)

### 21. Pacific Atlantic Water Flow
**Problem:** Find cells that can flow to both oceans.

```cpp
class Solution {
public:
    vector<vector<int>> pacificAtlantic(vector<vector<int>>& heights) {
        if (heights.empty()) return {};
        int m = heights.size(), n = heights[0].size();

        vector<vector<bool>> pacific(m, vector<bool>(n, false));
        vector<vector<bool>> atlantic(m, vector<bool>(n, false));

        // DFS from pacific edges
        for (int i = 0; i < m; i++) dfs(heights, pacific, i, 0);
        for (int j = 0; j < n; j++) dfs(heights, pacific, 0, j);

        // DFS from atlantic edges
        for (int i = 0; i < m; i++) dfs(heights, atlantic, i, n-1);
        for (int j = 0; j < n; j++) dfs(heights, atlantic, m-1, j);

        vector<vector<int>> result;
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (pacific[i][j] && atlantic[i][j]) {
                    result.push_back({i, j});
                }
            }
        }
        return result;
    }

private:
    void dfs(vector<vector<int>>& heights, vector<vector<bool>>& visited,
             int i, int j) {
        int m = heights.size(), n = heights[0].size();
        visited[i][j] = true;

        int dirs[4][2] = {{0,1}, {1,0}, {0,-1}, {-1,0}};
        for (auto& dir : dirs) {
            int x = i + dir[0], y = j + dir[1];
            if (x >= 0 && x < m && y >= 0 && y < n &&
                !visited[x][y] && heights[x][y] >= heights[i][j]) {
                dfs(heights, visited, x, y);
            }
        }
    }
};
```
**Time:** O(M*N), **Space:** O(M*N)

### 22. LRU Cache
**Problem:** Design LRU cache with O(1) operations.

```cpp
class LRUCache {
private:
    int capacity;
    list<pair<int, int>> cache;  // key, value
    unordered_map<int, list<pair<int, int>>::iterator> map;

public:
    LRUCache(int capacity) : capacity(capacity) {}

    int get(int key) {
        if (map.find(key) == map.end()) return -1;

        auto it = map[key];
        int value = it->second;
        cache.erase(it);
        cache.push_front({key, value});
        map[key] = cache.begin();
        return value;
    }

    void put(int key, int value) {
        if (map.find(key) != map.end()) {
            cache.erase(map[key]);
        } else if (cache.size() >= capacity) {
            int oldKey = cache.back().first;
            cache.pop_back();
            map.erase(oldKey);
        }

        cache.push_front({key, value});
        map[key] = cache.begin();
    }
};
```
**Time:** O(1) all operations, **Space:** O(capacity)

### 23. Sliding Window Maximum
**Problem:** Max in each sliding window.

```cpp
class Solution {
public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        deque<int> dq;  // Store indices
        vector<int> result;

        for (int i = 0; i < nums.size(); i++) {
            // Remove elements outside window
            while (!dq.empty() && dq.front() <= i - k) {
                dq.pop_front();
            }

            // Remove smaller elements (they won't be max)
            while (!dq.empty() && nums[dq.back()] < nums[i]) {
                dq.pop_back();
            }

            dq.push_back(i);

            if (i >= k - 1) {
                result.push_back(nums[dq.front()]);
            }
        }
        return result;
    }
};
```
**Time:** O(N), **Space:** O(k)

### 24. Maximum Frequency Stack
**Problem:** Stack where pop returns most frequent element.

```cpp
class FreqStack {
private:
    unordered_map<int, int> freq;
    unordered_map<int, stack<int>> freqStacks;
    int maxFreq = 0;

public:
    void push(int val) {
        freq[val]++;
        maxFreq = max(maxFreq, freq[val]);
        freqStacks[freq[val]].push(val);
    }

    int pop() {
        int val = freqStacks[maxFreq].top();
        freqStacks[maxFreq].pop();
        freq[val]--;

        if (freqStacks[maxFreq].empty()) {
            maxFreq--;
        }
        return val;
    }
};
```
**Time:** O(1) all operations, **Space:** O(N)

### 25. Serialize and Deserialize Binary Tree

```cpp
class Codec {
public:
    string serialize(TreeNode* root) {
        if (!root) return "#";
        return to_string(root->val) + "," + serialize(root->left) + "," +
               serialize(root->right);
    }

    TreeNode* deserialize(string data) {
        queue<string> nodes;
        stringstream ss(data);
        string node;
        while (getline(ss, node, ',')) {
            nodes.push(node);
        }
        return buildTree(nodes);
    }

private:
    TreeNode* buildTree(queue<string>& nodes) {
        string val = nodes.front();
        nodes.pop();

        if (val == "#") return nullptr;

        TreeNode* root = new TreeNode(stoi(val));
        root->left = buildTree(nodes);
        root->right = buildTree(nodes);
        return root;
    }
};
```
**Time:** O(N), **Space:** O(N)

### 26. Meeting Rooms II
**Problem:** Minimum conference rooms needed.

```cpp
class Solution {
public:
    int minMeetingRooms(vector<vector<int>>& intervals) {
        vector<int> start, end;
        for (auto& interval : intervals) {
            start.push_back(interval[0]);
            end.push_back(interval[1]);
        }

        sort(start.begin(), start.end());
        sort(end.begin(), end.end());

        int rooms = 0, endIdx = 0;
        for (int i = 0; i < start.size(); i++) {
            if (start[i] < end[endIdx]) {
                rooms++;
            } else {
                endIdx++;
            }
        }
        return rooms;
    }
};
```
**Time:** O(N log N), **Space:** O(N)

### 27. Find Anagrams in String
**Problem:** Find all anagram start indices.

```cpp
class Solution {
public:
    vector<int> findAnagrams(string s, string p) {
        vector<int> result;
        if (s.length() < p.length()) return result;

        vector<int> pCount(26, 0), sCount(26, 0);
        for (char c : p) pCount[c - 'a']++;

        for (int i = 0; i < s.length(); i++) {
            sCount[s[i] - 'a']++;

            if (i >= p.length()) {
                sCount[s[i - p.length()] - 'a']--;
            }

            if (i >= p.length() - 1 && sCount == pCount) {
                result.push_back(i - p.length() + 1);
            }
        }
        return result;
    }
};
```
**Time:** O(N), **Space:** O(1)
