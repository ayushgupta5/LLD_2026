### 1. Rolling Window Minimum Absolute Difference
**Problem:** For every K-sized window, find minimum absolute difference between any two elements.

**Optimal Approach: O(N log K)**
```cpp
class Solution {
public:
    vector<int> minAbsDiff(vector<int>& arr, int k) {
        vector<int> result;
        multiset<int> window;
        multiset<int> diffs;

        // Build first window
        for (int i = 0; i < k; i++) {
            auto it = window.insert(arr[i]);
            if (it != window.begin()) {
                auto prev = std::prev(it);
                diffs.insert(arr[i] - *prev);
            }
            auto next = std::next(it);
            if (next != window.end()) {
                diffs.insert(*next - arr[i]);
            }
        }
        result.push_back(*diffs.begin());

        // Slide window
        for (int i = k; i < arr.size(); i++) {
            int toRemove = arr[i - k];
            auto it = window.find(toRemove);

            // Remove differences involving toRemove
            if (it != window.begin()) {
                auto prev = std::prev(it);
                diffs.erase(diffs.find(toRemove - *prev));
            }
            auto next = std::next(it);
            if (next != window.end()) {
                diffs.erase(diffs.find(*next - toRemove));
                if (it != window.begin()) {
                    auto prev = std::prev(it);
                    diffs.insert(*next - *prev);
                }
            }
            window.erase(it);

            // Add new element
            it = window.insert(arr[i]);
            if (it != window.begin()) {
                auto prev = std::prev(it);
                diffs.insert(arr[i] - *prev);
                if (std::next(it) != window.end()) {
                    diffs.erase(diffs.find(*std::next(it) - *prev));
                }
            }
            if (std::next(it) != window.end()) {
                diffs.insert(*std::next(it) - arr[i]);
            }

            result.push_back(*diffs.begin());
        }
        return result;
    }
};
```
**Key Idea:** Maintain sorted window + sorted differences. O(log K) per operation.

### 2. Construct Array from Subsequences (Topological Sort)
**Problem:** Construct N-sized array [0..N-1] from K subsequences. Follow-up: Return lexicographically smallest.

**Approach: Topological Sort with DFS**
```cpp
class Solution {
    vector<vector<int>> adj;
    vector<bool> visited;
    vector<int> result;

    void dfs(int node) {
        visited[node] = true;
        // Sort in DESCENDING order (will reverse at end)
        sort(adj[node].rbegin(), adj[node].rend());
        for (int next : adj[node]) {
            if (!visited[next]) dfs(next);
        }
        result.push_back(node);
    }

public:
    vector<int> constructArray(int n, vector<vector<int>>& subsequences) {
        adj.resize(n);
        visited.resize(n, false);

        // Build graph
        for (auto& seq : subsequences) {
            for (int i = 0; i < seq.size() - 1; i++) {
                adj[seq[i]].push_back(seq[i + 1]);
            }
        }

        // DFS from all unvisited nodes (descending order for lex smallest)
        for (int i = n - 1; i >= 0; i--) {
            if (!visited[i]) dfs(i);
        }

        reverse(result.begin(), result.end());
        return result;
    }
};
```
**Key Idea:** Build dependency graph, DFS in descending order, reverse result.

### 3. Breaking Bad (Periodic Table Elements)
**Problem:** Match longest periodic symbols in a word. E.g., "henry" → "[He]nry" not "[H]enry"

**Optimal Approach: DP with Trie**
```cpp
struct TrieNode {
    unordered_map<char, TrieNode*> children;
    bool isEnd = false;
};

class Solution {
    TrieNode* root;

    void insert(string& s) {
        TrieNode* curr = root;
        for (char c : s) {
            char lower = tolower(c);
            if (!curr->children[lower])
                curr->children[lower] = new TrieNode();
            curr = curr->children[lower];
        }
        curr->isEnd = true;
    }

public:
    string breakingBad(string name, vector<string>& symbols) {
        root = new TrieNode();
        int maxLen = 0;
        for (string& sym : symbols) {
            insert(sym);
            maxLen = max(maxLen, (int)sym.size());
        }

        vector<string> words;
        stringstream ss(name);
        string word;
        while (ss >> word) words.push_back(word);

        string result;
        for (string& w : words) {
            if (!result.empty()) result += " ";
            result += processWord(w, maxLen);
        }
        return result;
    }

    string processWord(string& word, int maxLen) {
        int n = word.size();
        vector<int> dp(n + 1, -1); // dp[i] = longest match ending before i
        vector<int> matchLen(n + 1, 0);
        dp[0] = 0;

        for (int i = 0; i < n; i++) {
            if (dp[i] == -1) continue;
            TrieNode* curr = root;

            for (int j = i; j < min(i + maxLen, n); j++) {
                char c = tolower(word[j]);
                if (!curr->children[c]) break;
                curr = curr->children[c];

                if (curr->isEnd) {
                    int len = j - i + 1;
                    if (dp[j + 1] == -1 || len > matchLen[j + 1]) {
                        dp[j + 1] = i;
                        matchLen[j + 1] = len;
                    }
                }
            }
        }

        // Reconstruct
        string ans;
        int pos = n;
        while (pos > 0) {
            if (matchLen[pos] > 0) {
                int start = dp[pos];
                string match = word.substr(start, matchLen[pos]);
                ans = "[" + match + "]" + ans;
                pos = start;
            } else {
                ans = word[pos - 1] + ans;
                pos--;
            }
        }
        return ans;
    }
};
```
**Key Idea:** DP + Trie. Track longest symbol match at each position.

### 4. Haunted House Problem
**Problem:** N people with constraints [Li, Ri]. Person i goes with Li to Ri others. Max group size?

**Optimal Approach: O(N)**
```cpp
class Solution {
public:
    int maxPeople(vector<pair<int,int>>& constraints) {
        int n = constraints.size();
        int maxAns = 0;

        // Try each possible group size
        for (int groupSize = 1; groupSize <= n; groupSize++) {
            int count = 0;
            for (auto& [l, r] : constraints) {
                int others = groupSize - 1;
                if (others >= l && others <= r) count++;
            }
            if (count >= groupSize) {
                maxAns = groupSize;
            }
        }
        return maxAns;
    }
};
```
**Key Idea:** For group size K, need at least K people satisfied with K-1 others.

### 5. Number of Islands with Dynamic Updates
**Problem:** `setLand(i,j)`, `isLand(i,j)`, `numberOfIslands()` with frequent queries.

**Optimal Approach: Union-Find**
```cpp
class UnionFind {
    unordered_map<int, int> parent;
    unordered_map<int, int> rank;
public:
    int count = 0;

    int find(int x) {
        if (!parent.count(x)) {
            parent[x] = x;
            rank[x] = 1;
            count++;
            return x;
        }
        if (parent[x] != x) parent[x] = find(parent[x]);
        return parent[x];
    }

    void unite(int x, int y) {
        int px = find(x), py = find(y);
        if (px == py) return;
        if (rank[px] < rank[py]) swap(px, py);
        parent[py] = px;
        if (rank[px] == rank[py]) rank[px]++;
        count--;
    }

    bool exists(int x) {
        return parent.count(x);
    }
};

class IslandTracker {
    set<pair<int,int>> land;
    UnionFind uf;

    int encode(int i, int j) { return i * 100000 + j; }

public:
    void setLand(int i, int j) {
        if (land.count({i, j})) return;
        land.insert({i, j});

        int curr = encode(i, j);
        uf.find(curr);

        int dirs[4][2] = {{0,1},{0,-1},{1,0},{-1,0}};
        for (auto& d : dirs) {
            int ni = i + d[0], nj = j + d[1];
            if (land.count({ni, nj})) {
                uf.unite(curr, encode(ni, nj));
            }
        }
    }

    bool isLand(int i, int j) {
        return land.count({i, j});
    }

    int numberOfIslands() {
        return uf.count;
    }
};
```
**Key Idea:** Union-Find with path compression. O(1) for count, O(α(n)) for updates.

### 6. Making a Large Island (LC 827)
**Problem:** Grid of 0s and 1s. Change one 0 to 1 to maximize island size.

```cpp
class Solution {
    int dfs(vector<vector<int>>& grid, int i, int j, int id) {
        if (i < 0 || i >= grid.size() || j < 0 || j >= grid[0].size()
            || grid[i][j] != 1) return 0;
        grid[i][j] = id;
        return 1 + dfs(grid, i+1, j, id) + dfs(grid, i-1, j, id)
                 + dfs(grid, i, j+1, id) + dfs(grid, i, j-1, id);
    }

public:
    int largestIsland(vector<vector<int>>& grid) {
        int n = grid.size();
        unordered_map<int, int> area;
        int id = 2, maxArea = 0;

        // Label islands and compute areas
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (grid[i][j] == 1) {
                    area[id] = dfs(grid, i, j, id);
                    maxArea = max(maxArea, area[id]);
                    id++;
                }
            }
        }

        // Try flipping each 0
        int dirs[4][2] = {{0,1},{0,-1},{1,0},{-1,0}};
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (grid[i][j] == 0) {
                    unordered_set<int> neighbors;
                    for (auto& d : dirs) {
                        int ni = i + d[0], nj = j + d[1];
                        if (ni >= 0 && ni < n && nj >= 0 && nj < n
                            && grid[ni][nj] > 1) {
                            neighbors.insert(grid[ni][nj]);
                        }
                    }
                    int size = 1;
                    for (int id : neighbors) size += area[id];
                    maxArea = max(maxArea, size);
                }
            }
        }
        return maxArea;
    }
};
```

### 7. Find Median from Data Stream (LC 295)

```cpp
class MedianFinder {
    priority_queue<int> maxHeap; // left half
    priority_queue<int, vector<int>, greater<int>> minHeap; // right half

public:
    void addNum(int num) {
        maxHeap.push(num);
        minHeap.push(maxHeap.top());
        maxHeap.pop();

        if (maxHeap.size() < minHeap.size()) {
            maxHeap.push(minHeap.top());
            minHeap.pop();
        }
    }

    double findMedian() {
        if (maxHeap.size() > minHeap.size()) return maxHeap.top();
        return (maxHeap.top() + minHeap.top()) / 2.0;
    }
};
```

**Follow-up: Loose Median (Range between powers of 2)**
```cpp
// Use buckets: O(1) time, O(log MaxVal) space
class LooseMedianFinder {
    map<int, int> buckets; // bucket_id -> count
    int total = 0;

    int getBucket(int num) {
        if (num <= 0) return 0;
        return log2(num);
    }

public:
    void addNum(int num) {
        buckets[getBucket(num)]++;
        total++;
    }

    int findLooseMedian() {
        int target = (total + 1) / 2;
        int count = 0;
        for (auto& [bucket, cnt] : buckets) {
            count += cnt;
            if (count >= target) {
                // Return any value in range [2^bucket, 2^(bucket+1))
                return (1 << bucket);
            }
        }
        return 0;
    }
};
```

### 8. Binary Tree Right Side View (LC 199)

```cpp
vector<int> rightSideView(TreeNode* root) {
    vector<int> result;
    if (!root) return result;

    queue<TreeNode*> q;
    q.push(root);

    while (!q.empty()) {
        int size = q.size();
        for (int i = 0; i < size; i++) {
            TreeNode* node = q.front(); q.pop();
            if (i == size - 1) result.push_back(node->val);
            if (node->left) q.push(node->left);
            if (node->right) q.push(node->right);
        }
    }
    return result;
}
```

```cpp
class Solution {
    bool dfs(vector<vector<char>>& board, string& word, int i, int j, int idx) {
        if (idx == word.size()) return true;
        if (i < 0 || i >= board.size() || j < 0 || j >= board[0].size()
            || board[i][j] != word[idx]) return false;

        char temp = board[i][j];
        board[i][j] = '#';

        bool found = dfs(board, word, i+1, j, idx+1) ||
                     dfs(board, word, i-1, j, idx+1) ||
                     dfs(board, word, i, j+1, idx+1) ||
                     dfs(board, word, i, j-1, idx+1);

        board[i][j] = temp;
        return found;
    }

public:
    bool exist(vector<vector<char>>& board, string word) {
        for (int i = 0; i < board.size(); i++) {
            for (int j = 0; j < board[0].size(); j++) {
                if (dfs(board, word, i, j, 0)) return true;
            }
        }
        return false;
    }
};
```

```cpp
            }
        }

    return true;

class Solution {
    bool solve(vector<vector<char>>& board) {
        for (int i = 0; i < 9; i++) {
            for (int j = 0; j < 9; j++) {
                if (board[i][j] == '.') {
                    for (char c = '1'; c <= '9'; c++) {
                        if (isValid(board, i, j, c)) {
                            board[i][j] = c;
                            if (solve(board)) return true;
                            board[i][j] = '.';
                        }
                    }
                    return false;
                }
            }
        }
        return true;
    }

            if (board[row][i] == c || board[i][col] == c ||
                board[row/3*3 + i/3][col/3*3 + i%3] == c) return false;
        }
        return true;
    }

public:
    void solveSudoku(vector<vector<char>>& board) {
        solve(board);
    }
};
```

### 12. Find Nearest Palindrome (LC 564)

```cpp
class Solution {
public:
    string nearestPalindromic(string n) {
        int len = n.size();
        long long num = stoll(n);
        set<long long> candidates;

        // Edge cases
        candidates.insert((long long)pow(10, len) + 1); // 1000...001
        candidates.insert((long long)pow(10, len - 1) - 1); // 999...999

        // Mirror prefix
        long long prefix = stoll(n.substr(0, (len + 1) / 2));
        for (int i = -1; i <= 1; i++) {
            string p = to_string(prefix + i);
            string palin = p + string(p.rbegin() + (len % 2), p.rend());
            candidates.insert(stoll(palin));
        }

        candidates.erase(num);

        long long result = 0, minDiff = LLONG_MAX;
        for (long long cand : candidates) {
            long long diff = abs(cand - num);
            if (diff < minDiff || (diff == minDiff && cand < result)) {
                minDiff = diff;
                result = cand;
            }
        }
        return to_string(result);
    }
};
```

### 13. Alien Dictionary (LC 269)

```cpp
string alienOrder(vector<string>& words) {
    unordered_map<char, unordered_set<char>> graph;
    unordered_map<char, int> indegree;

    // Initialize
    for (string& w : words)
        for (char c : w) indegree[c] = 0;

    // Build graph
    for (int i = 0; i < words.size() - 1; i++) {
        string w1 = words[i], w2 = words[i + 1];
        int minLen = min(w1.size(), w2.size());

        if (w1.size() > w2.size() && w1.substr(0, minLen) == w2)
            return ""; // Invalid

        for (int j = 0; j < minLen; j++) {
            if (w1[j] != w2[j]) {
                if (!graph[w1[j]].count(w2[j])) {
                    graph[w1[j]].insert(w2[j]);
                    indegree[w2[j]]++;
                }
                break;
            }
        }
    }

    // Topological sort (Kahn's algorithm)
    queue<char> q;
    for (auto& [c, deg] : indegree)
        if (deg == 0) q.push(c);

    string result;
    while (!q.empty()) {
        char curr = q.front(); q.pop();
        result += curr;
        for (char next : graph[curr]) {
            if (--indegree[next] == 0) q.push(next);
        }
    }

    return result.size() == indegree.size() ? result : "";
}
```

```cpp
string base2To6(string binary) {
    // Convert binary to decimal
    long long decimal = 0;
    for (char c : binary) {
        decimal = decimal * 2 + (c - '0');
    }

    // Convert decimal to base 6
    if (decimal == 0) return "0";
    string result;
    while (decimal > 0) {
        result = to_string(decimal % 6) + result;
        decimal /= 6;
    }
    return result;
```

```java
        } finally {
```

### 2. HashMap with getRandom()

```java
class RandomizedMap<K, V> {
    private Map<K, V> map;
    private List<K> keys;
    private Map<K, Integer> keyToIndex;
    private Random random;
    private ReentrantReadWriteLock lock;

    public RandomizedMap() {
        map = new HashMap<>();
        keys = new ArrayList<>();
        keyToIndex = new HashMap<>();
        random = new Random();
        lock = new ReentrantReadWriteLock();
    }

    public void put(K key, V value) {
        lock.writeLock().lock();
        try {
            if (!map.containsKey(key)) {
                keys.add(key);
                keyToIndex.put(key, keys.size() - 1);
            }
            map.put(key, value);
        } finally {
            lock.writeLock().unlock();
        }
    }

    public V get(K key) {
        lock.readLock().lock();
        try {
            return map.get(key);
        } finally {
            lock.readLock().unlock();
        }
    }

    public boolean remove(K key) {
        lock.writeLock().lock();
        try {
            if (!map.containsKey(key)) return false;

            int index = keyToIndex.get(key);
            K lastKey = keys.get(keys.size() - 1);

            // Swap with last
            keys.set(index, lastKey);
            keyToIndex.put(lastKey, index);

            // Remove last
            keys.remove(keys.size() - 1);
            keyToIndex.remove(key);
            map.remove(key);
            return true;
        } finally {
            lock.writeLock().unlock();
        }
    }

    public Pair<K, V> getRandom() {
        lock.readLock().lock();
        try {
            K randomKey = keys.get(random.nextInt(keys.size()));
            return new Pair<>(randomKey, map.get(randomKey));
        } finally {
            lock.readLock().unlock();
        }
    }
}
```

**Key Classes:**
- `RandomizedMap<K,V>`
- Uses `HashMap<K,V>`, `ArrayList<K>`, `HashMap<K,Integer>`
- Thread-safe with `ReentrantReadWriteLock`

### 5. Leaderboard

```java
class Player {
    String id;
    int score;
}

class Leaderboard {
    private TreeMap<Integer, Set<String>> scoreToPlayers; // score -> player IDs
    private Map<String, Integer> playerToScore;
    private ReentrantReadWriteLock lock;

    // Cache
    private List<Player> cachedTop100;
    private long cacheTimestamp;
    private static final long CACHE_TTL = 60000; // 1 minute

    public void updateScore(String playerId, int newScore) {
        lock.writeLock().lock();
        try {
            // Remove old score
            if (playerToScore.containsKey(playerId)) {
                int oldScore = playerToScore.get(playerId);
                scoreToPlayers.get(oldScore).remove(playerId);
                if (scoreToPlayers.get(oldScore).isEmpty()) {
                    scoreToPlayers.remove(oldScore);
                }
            }

            // Add new score
            playerToScore.put(playerId, newScore);
            scoreToPlayers.putIfAbsent(newScore, new HashSet<>());
            scoreToPlayers.get(newScore).add(playerId);

            // Invalidate cache
            cachedTop100 = null;
        } finally {
            lock.writeLock().unlock();
        }
    }

    public List<Player> getTopN(int n) {
        lock.readLock().lock();
        try {
            if (cachedTop100 != null && System.currentTimeMillis() - cacheTimestamp < CACHE_TTL) {
                return cachedTop100.subList(0, Math.min(n, cachedTop100.size()));
            }
        } finally {
            lock.readLock().unlock();
        }

        // Recompute cache
        lock.writeLock().lock();
        try {
            List<Player> result = new ArrayList<>();
            for (Map.Entry<Integer, Set<String>> entry : scoreToPlayers.descendingMap().entrySet()) {
                for (String id : entry.getValue()) {
                    result.add(new Player(id, entry.getKey()));
                    if (result.size() >= 100) break;
                }
                if (result.size() >= 100) break;
            }
            cachedTop100 = result;
            cacheTimestamp = System.currentTimeMillis();
            return result.subList(0, Math.min(n, result.size()));
        } finally {
            lock.writeLock().unlock();
        }
    }
}

// Thread-safe with Keyed Executor
class KeyedExecutor {
    private Map<String, Lock> locks = new ConcurrentHashMap<>();

    public void execute(String key, Runnable task) {
        Lock lock = locks.computeIfAbsent(key, k -> new ReentrantLock());
        lock.lock();
        try {
            task.run();
        } finally {
            lock.unlock();
        }
    }
}
```
